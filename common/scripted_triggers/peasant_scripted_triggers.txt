is_blocked_from_peasant_government_because_of_treaty = {
	custom_trigger_tooltip = {
		tooltip = is_blocked_from_peasant_government_because_of_treaty_tt
		has_country_flag = cannot_have_peasant_government_because_of_treaty
		NOT = {
			had_country_flag = {
				flag = cannot_have_peasant_government_because_of_treaty
				days = 7300
			}
		}
	}
}

country_or_commune_subject_or_subject_of_commune_subject_holds = {
	custom_trigger_tooltip = {
		tooltip = country_or_commune_subject_or_subject_of_commune_subject_holds_tt
		OR = {
			owned_by = $who$
			owner = {
				is_commune_subject = yes
				is_subject_of = $who$
			}
			owner = {
				is_subject = yes
				overlord = {
					is_commune_subject = yes
					is_subject_of = $who$
				}
			}
		}
	}
}
