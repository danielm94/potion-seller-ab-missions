#SCOPE = COUNTRY
is_commune_subject = {
	OR = {
		is_subject_of_type = allied_commune
		is_subject_of_type = sister_commune
		is_subject_of_type = integrated_commune
	}
}

#SCOPE = COUNTRY

	has_country_flag = commune_is_being_integrated_subject
}

#SCOPE = COUNTRY

	has_country_flag = commune_is_being_integrated_overlord
}

#SCOPE = COUNTRY
is_commune_integration_progress_greater_or_equal_to = {
	variable_arithmetic_trigger = {
		export_to_variable = {
			which = curr_integration_progress
			value = trigger_value:always
		}
		change_variable = {
			which = curr_integration_progress
			which = commune_integration_progress
		}
		divide_variable = {
			which = curr_integration_progress
			which = commune_integration_cost
		}
		check_variable = {
			curr_integration_progress = $decimal_percentage$
		}
	}
}
