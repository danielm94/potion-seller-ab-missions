#Checks if targetA has more prestige than targetB
has_more_prestige_than = {
	variable_arithmetic_trigger = {
		[[tooltip]
		custom_tooltip = $tooltip$
		]
		export_to_variable = {
			which = target_a_prestige
			value = prestige
			who = $targetA$
		}
		export_to_variable = {
			which = target_b_prestige
			value = prestige
			who = $targetB$
		}
		NOT = {
			check_variable = {
				which = target_b_prestige
				which = target_a_prestige
			}
		}
	}
}

has_more_total_development_than_target = {
	$target$ = {
		variable_arithmetic_trigger = {
			custom_tooltip = more_development_than_target_tt
			export_to_variable = {
				which = target_total_dev
				value = total_development
				who = $target$
			}
			export_to_variable = {
				which = root_total_dev
				value = total_development
				who = ROOT
			}
			check_variable = {
				which = root_total_dev
				which = target_total_dev
			}
		}
	}
}

# Checks if you are currently supporting rebels before they rise up against the target.
is_supporting_rebels_against_target = {
	$target$ = {
		custom_trigger_tooltip = {
			tooltip = is_supporting_rebels_against_target_tt
			OR = {
				AND = {
					has_country_flag = support_rebels_against_@THIS
					ROOT = {
						has_country_flag = support_rebels_against_@PREV
					}
				}
				ROOT = {
					has_spawned_supported_rebels = $target$
				}
			}
		}
	}
}
