is_variable_less_than_or_equal_to = {
	OR = {
		is_variable_equal = {
			which = $var$
			[[value]
			value = $value$
			]
			[[which]
			which = $which$
			]
		}
		NOT = {
			check_variable = {
				which = $var$
				[[value]
				value = $value$
				]
				[[which]
				which = $which$
				]
			}
		}
	}
}
