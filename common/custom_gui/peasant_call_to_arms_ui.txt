custom_button = {
	name = peasant_call_to_arms_ally
	tooltip = peasant_call_to_arms_ally_tt
	potential = {
		is_commune_subject = yes
		FROM = {
			is_at_war = yes
			has_government_mechanic = peasant_zealotry
		}
	}
	trigger = {
		if = {
			limit = {
				FROM = {
					any_war_enemy_country = {
						NOT = {
							truce_with = ROOT
						}
						NOT = {
							war_with = ROOT
						}
						NOT = {
							is_subject_of = ROOT
						}
						OR = {
							is_in_war = {
								defender_leader = THIS
								attackers = FROM
							}
							is_in_war = {
								attacker_leader = THIS
								defenders = FROM
							}
						}
						NOT = {
							any_known_country = {
								is_in_war = {
									defender_leader = PREV
									attackers = FROM
									defenders = THIS
								}
								OR = {
									truce_with = ROOT
									war_with = ROOT
									is_subject_of = ROOT
								}
							}
						}
						NOT = {
							any_known_country = {
								is_in_war = {
									attacker_leader = PREV
									defenders = FROM
									attackers = THIS
								}
								OR = {
									truce_with = ROOT
									war_with = ROOT
									is_subject_of = ROOT
								}
							}
						}
					}
				}
			}
			can_call_this_commune_to_war = yes
			custom_trigger_tooltip = {
				tooltip = peasant_call_to_arms_trigger_tt
				always = yes
			}
		}
		else = {
			custom_trigger_tooltip = {
				tooltip = peasant_call_to_arms_trigger_tt
				always = no
			}
		}
		NOT = {
			liberty_desire = 50
		}
	}
	effect = {
		if = {
			limit = {
				check_variable = {
					join_wars_peasant_zeal_cost = 0.001
				}
			}
			custom_tooltip = peasant_call_to_arms_ally_effect_tt
		}
		hidden_effect = {
			join_all_offensive_wars_of = from
			join_all_defensive_wars_of = from
			FROM = {
				set_variable = {
					which = join_wars_peasant_zeal_cost
					which = ROOT
				}
				multiply_variable = {
					which = join_wars_peasant_zeal_cost
					value = -1
				}
				variable_government_power_effect = {
					var = join_wars_peasant_zeal_cost
					mechanic_type = peasant_zealotry
					power_type = peasant_zeal
					scaled_with_gain_modifier = no
				}
				every_subject_country = {
					limit = {
						is_commune_subject = yes
					}
					calculate_peasant_zeal_call_to_arms_cost = yes
				}
			}
		}
	}
}

custom_button = {
	name = peasant_call_to_arms_enemy
	tooltip = peasant_call_to_arms_enemy_tt
	potential = {
		FROM = {
			has_government_mechanic = peasant_zealotry
		}
		OR = {
			is_in_war = {
				attacker_leader = root
				defenders = from
			}
			is_in_war = {
				defender_leader = root
				attackers = from
			}
		}
	}
	trigger = {
		FROM = {
			any_subject_country = {
				custom_trigger_tooltip = {
					tooltip = is_a_commune_subject_tt
					is_commune_subject = yes
				}
				if = {
					limit = {
						PREV = {
							NOT = {
								has_country_flag = cta_allied_communes
							}
						}
					}
					NOT = {
						is_subject_of_type = allied_commune
					}
				}
				if = {
					limit = {
						PREV = {
							NOT = {
								has_country_flag = cta_integrated_communes
							}
						}
					}
					NOT = {
						is_subject_of_type = integrated_commune
					}
				}
				NOT = {
					liberty_desire = 50
				}
				NOT = {
					war_with = ROOT
				}
				NOT = {
					truce_with = ROOT
				}
				custom_trigger_tooltip = {
					tooltip = any_subject_doesnt_have_truce_trigger_tt
					NOT = {
						any_country = {
							OR = {
								is_in_war = {
									attacker_leader = ROOT
									defenders = FROM
									attackers = THIS
								}
								is_in_war = {
									defender_leader = ROOT
									attackers = FROM
									defenders = THIS
								}
							}
							OR = {
								truce_with = PREV
								war_with = PREV
								is_subject_of = PREV
							}
						}
					}
				}
				custom_trigger_tooltip = {
					tooltip = cost_to_call_less_than_budget_tt
					has_peasant_zeal_cost_less_or_equal_to_budget = yes
				}
			}
		}
		custom_trigger_tooltip = {
			tooltip = peasant_call_to_arms_enemy_highest_zeal_tt
			event_target:highest_coa_cost = {
				can_call_this_commune_to_war = yes
			}
		}
	}
	effect = {
		custom_tooltip = tooltip_line_divider
		custom_tooltip = peasant_call_to_arms_enemy_effect_tt
		custom_tooltip = AB_EMPTY
		if = {
			limit = {
				FROM = {
					has_country_flag = peasant_call_to_arms_strongest_military_strategy
				}
			}
			custom_tooltip = peasant_call_to_arms_strongest_military_strategy_tt
		}
		else_if = {
			limit = {
				FROM = {
					has_country_flag = peasant_call_to_arms_closest_to_target_strategy
				}
			}
			custom_tooltip = peasant_call_to_arms_closest_to_target_strategy_tt
		}
		custom_tooltip = AB_EMPTY
		custom_tooltip = call_all_communes_type_tt
		if = {
			limit = {
				FROM = {
					has_country_flag = cta_allied_communes
				}
			}
			custom_tooltip = cta_allied_communes_yes_tt
		}
		else = {
			custom_tooltip = cta_allied_communes_no_tt
		}
		if = {
			limit = {
				FROM = {
					has_country_flag = cta_integrated_communes
				}
			}
			custom_tooltip = cta_integrated_communes_yes_tt
		}
		else = {
			custom_tooltip = cta_integrated_communes_no_tt
		}
		custom_tooltip = AB_EMPTY
		custom_tooltip = peasant_zeal_budget_tt
		hidden_effect = {
			FROM = {
				every_subject_country = {
					limit = {
						is_commune_subject = yes
					}
					calculate_peasant_zeal_call_to_arms_cost = yes
				}
				call_all_peasant_subjects_to_arms_effect = yes
			}
		}
	}
}
