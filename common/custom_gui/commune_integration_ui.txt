custom_button = {
	name = integrate_commune
	tooltip = integrate_commune
	potential = {
		FROM = {
			has_government_mechanic = peasant_zealotry
		}
		is_subject_of_type = allied_commune
		is_commune_subject_being_integrated = no
	}
	trigger = {
		if = {
			limit = {
				has_country_flag = commune_integrated_recently
				NOT = {
					had_country_flag = {
						flag = commune_integrated_recently
						days = 3650
					}
				}
			}
			custom_trigger_tooltip = {
				tooltip = integrated_commune_recently_tt
				always = no
			}
		}
		else = {
			FROM = {
				num_of_diplomats = 2
			}
			is_commune_loyal_enough_to_integrate = yes
			are_there_no_truces_with_target_diplomacy_gui = {
				target = FROM
			}
			has_opinion = {
				who = FROM
				value = 190
			}
			trust = {
				who = FROM
				value = 50
			}
		}
	}
	effect = {
		hidden_effect = {
			FROM = {
				play_sound = gain_prestige
			}
		}
		custom_tooltip = tooltip_line_divider
		custom_tooltip = integrate_commune_desc
		hidden_effect = {
			FROM = {
				enable_commune_integration_overlord = yes
			}
			initialize_integration_progress_percentage_variable = yes
			enable_commune_integration_subject = yes
			set_country_flag = investing_dip_in_commune
			FROM = {
				push_commune_subject_investing_mana_modifier = yes
			}
			calculate_commune_integration_cost = yes
			calculate_monthly_commune_integration_progress = yes
			calculate_integration_progress_percentage = yes
			ROOT = {
				FROM = {
					calculate_overlord_dip_investment_integration_progress_gain = yes
				}
			}
			FROM = {
				map_all_integrating_commune_subjects_to_slots = yes
			}
		}
	}
}

custom_button = {
	name = cancel_integrate_commune
	tooltip = cancel_commune_integration
	potential = {
		FROM = {
			has_government_mechanic = peasant_zealotry
		}
		is_commune_subject_being_integrated = yes
	}
	trigger = {
	}
	effect = {
		hidden_effect = {
			FROM = {
				play_sound = back_click
			}
		}
		custom_tooltip = cancel_commune_integration_desc
		hidden_effect = {
			FROM = {
				if = {
					limit = {
						NOT = {
							any_subject_country = {
								is_commune_subject_being_integrated = yes
							}
						}
					}
					disable_commune_integration_overlord = yes
				}
			}
			disable_commune_integration_subject = yes
			initialize_commune_integration_progress = yes
			initialize_integration_progress_percentage_variable = yes
			clear_commune_integration_flags = yes
			FROM = {
				pop_commune_subject_investing_mana_modifier = yes
			}
			calculate_monthly_commune_integration_progress = yes
			FROM = {
				map_all_integrating_commune_subjects_to_slots = yes
			}
		}
	}
}

custom_button = {
	name = stage_coup
	tooltip = stage_coup_commune_integration
	potential = {
		FROM = {
			has_government_mechanic = peasant_zealotry
		}
		is_subject_of = FROM
		is_commune_subject_being_integrated = yes
		is_commune_subject = yes
	}
	trigger = {
		FROM = {
			custom_trigger_tooltip = {
				tooltip = stage_coup_spy_network_size_tt
				can_stage_coup = {
					who = ROOT
				}
			}
			variable_arithmetic_trigger = {
				custom_tooltip = has_enough_ducats_for_coup_tt
				export_to_variable = {
					which = target_dev
					value = total_development
					who = ROOT
				}
				export_to_variable = {
					which = ducats
					value = treasury
				}
				check_variable = {
					which = ducats
					which = target_dev
				}
			}
		}
		custom_trigger_tooltip = {
			tooltip = coup_leader_still_alive_tt
			NOT = {
				has_ruler_flag = commune_coup_leader
			}
		}
	}
	effect = {
		hidden_effect = {
			FROM = {
				play_sound = revolutionary_republic
			}
		}
		custom_tooltip = tooltip_line_divider
		custom_tooltip = stage_coup_commune_integration_desc
		hidden_effect = {
			FROM = {
				change_variable = {
					total_staged_coups_against_commune = 1
				}
				calculate_spy_action_cost_values = yes
				set_variable = {
					which = spy_network_deduction_var
					which = stage_coup_cost_var
				}
				multiply_variable = {
					which = spy_network_deduction_var
					value = -1
				}
				variable_change_spy_network_in = {
					who = ROOT
					var = spy_network_deduction_var
				}
				export_to_variable = {
					which = total_dev
					value = total_development
					who = ROOT
				}
				multiply_variable = {
					total_dev = -1
				}
				variable_effect = {
					var = total_dev
					effect = add_treasury
				}
			}
			kill_ruler = yes
			set_ruler_flag = commune_coup_leader
			change_commune_integration_progress_as_percentage_of_cost = {
				percentage = 0.2
			}
			calculate_integration_progress_percentage = yes
			add_liberty_desire = 40
			add_opinion = {
				who = FROM
				modifier = commune_staged_coup_against_us
			}
			add_trust = {
				who = FROM
				value = -40
			}
			FROM = {
				every_subject_country = {
					limit = {
						NOT = {
							tag = ROOT
						}
					}
					add_liberty_desire = 20
					add_trust = {
						who = FROM
						value = -20
					}
					add_opinion = {
						who = FROM
						modifier = commune_staged_coup_against_another_commune
					}
				}
			}
		}
	}
}
