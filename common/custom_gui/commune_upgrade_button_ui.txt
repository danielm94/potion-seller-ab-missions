custom_button = {
	name = kiev_commune_upgrade_button
	tooltip = construct_great_commune_tt
	potential = {
		province_id = 280
		has_province_modifier = kievan_commune
		owner = {
			has_government_mechanic = peasant_zealotry
		}
		owned_by = FROM
	}
	trigger = {
		if = {
			limit = {
				owner = {
					is_debug_mode_enabled = yes
				}
			}
			always = yes
		}
		else = {
			development = 30
			num_of_times_expanded_infrastructure = 2
			has_great_project = {
				type = kiev_pechersk_lavra
				tier = 1
			}
			has_tax_building_trigger = yes
			has_manpower_building_trigger = yes
			has_fort_building_trigger = yes
		}
	}
	effect = {
		custom_tooltip = tooltip_line_divider
		remove_province_modifier = kievan_commune
		add_province_modifier = {
			name = great_kievan_commune
			duration = -1
		}
		if = {
			limit = {
				has_great_project = {
					type = kiev_pechersk_lavra
					tier = 3
				}
			}
			owner = {
				add_treasury = 3000
			}
		}
		else = {
			add_great_project_tier = {
				type = kiev_pechersk_lavra
				tier = 1
			}
		}
		custom_tooltip = AB_EMPTY
		owner = {
			define_advisor = {
				type = patriarch
				skill = 2
				cost_multiplier = 0.5
				culture = ruthenian
				religion = orthodox
				age = 40
				location = 280
			}
			add_patriarch_authority = 0.1
			add_faction_influence = {
				faction = exemplarists_faction
				influence = 10
			}
		}
	}
}
