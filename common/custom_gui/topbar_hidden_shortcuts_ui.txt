custom_button = {
	name = toggle_verbose_tooltips_hidden_button
	tooltip = AB_EMPTY
	potential = {
		FROM = {
			OR = {
				has_country_flag = in_peasant_elections_event
				has_country_flag = in_moldavia_peasant_events.7
			}
		}
	}
	trigger = {
	}
	effect = {
		FROM = {
			if = {
				limit = {
					has_country_flag = in_peasant_elections_event
				}
				toggle_country_flag_with_sound = {
					flag = hide_peasant_election_verbose_tooltips
				}
			}
			else_if = {
				limit = {
					has_country_flag = in_moldavia_peasant_events.7
				}
				toggle_country_flag_with_sound = {
					flag = show_verbose_gpw_tooltips
				}
			}
		}
	}
}
