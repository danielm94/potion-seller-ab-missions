	name = support_rebels_config_button
	tooltip = support_rebels_config_button_tt
	potential = {
		OR = {
			AND = {
				has_country_flag = support_rebels_against_@ROOT
				FROM = {
					has_country_flag = support_rebels_against_@ROOT
				}
			}
			FROM = {
				has_spawned_supported_rebels = PREV
			}
		}
	}
	trigger = {
		if = {
			limit = {
				FROM = {
					has_country_flag = in_support_rebels_config_events.0
				}
			}
			custom_trigger_tooltip = {
				tooltip = not_in_support_rebels_config_events.0_tt
				always = no
			}
		}
	}
	effect = {
		FROM = {
			country_event = {
				id = support_rebels_config_events.0
			}
		}
	}
}
