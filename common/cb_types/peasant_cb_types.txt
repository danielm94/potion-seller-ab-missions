			has_country_flag = incited_peasants
			NOT = {
				had_country_flag = {
					flag = incited_peasants
					days = 3650
				}
			}
			NOT = {

			}
		}
	}
	war_goal = overthrow_the_nobility_wargoal
}

cb_crush_peasants = {
	valid_for_subject = no
	ai_peace_desire = -30
	prerequisites_self = {
		NOT = {
			has_government_attribute = is_peasants_republic
		}
	}
	prerequisites = {
		#prevent this CB from messing with great peasant war mechanics...
		if = {
			limit = {
				FROM = {
					is_part_of_hre = yes
				}
			}
			is_part_of_hre = no
		}
		FROM = {
			has_government_attribute = is_peasants_republic
		}
		OR = {
			government = monarchy
			government = theocracy
			AND = {
				government = republic
				FROM = {
					OR = {
						NOT = {
							stability = 0
						}
						has_government_power = {
							mechanic_type = peasant_zealotry
							power_type = peasant_zeal
							value = 50
						}
					}
				}
			}
		}
		is_neighbor_of = FROM
	}
	war_goal = crush_peasants_wargoal
}
