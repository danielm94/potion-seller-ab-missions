po_force_consulate = {
	# Show in treaties tab
	category = 6
	prestige_base = 0.5
	ae_base = 0.5
	power_projection = vassalized_rival
	power_cost_base = 0
	# Warscore cost scales directly according to the warscore cost of the provinces that the owner retains
	# IE: If you take some provinces, the cost of forced conversion is reduced
	warscore_cost = {
		all_provinces = 1
		owner_keeps = 0		# The provinces that are not of any of the types below
		conquered = 0		# Provinces that are selected by a Demand Province treaty
		returned_core = 0		# Provinces that are selected by a Return Core treaty
		released_nation = 0		# Provinces that will become a part of a selected released-to-be nation
		cancelled_subject = 0		# This one should not be relevant here since Giver can't be a cancelled subject. Including anyway for completeness
		concede_colonial_region = 0		# Ditto
	}
	# No limit to the cost of forced conversion (ie. large nations cannot be forcibly converted)
	warscore_cap = 100
	requires_is_allowed = no
	is_make_subject = no
	applies_to_war_target = yes
	is_visible = {
		is_in_war = {
			defender_leader = FROM
			attacker_leader = ROOT
			casus_belli = cb_trade_conflict
		}
		has_country_flag = enable_agitate_guilds
		FROM = {
			has_country_flag = agitated_guilds
		}
	}
	is_allowed = {
		always = yes
	}
	effect = {
		create_subject = {
			subject_type = consulate
			who = FROM
		}
		FROM = {
			kill_consort = yes
			kill_heir = {
				allow_new_heir = no
			}
			kill_ruler = yes
			if = {
				limit = {
					NOT = {
						has_government_attribute = is_merchant_republic
					}
				}
				change_government = republic
				add_government_reform = merchants_reform
			}
			remove_country_modifier = agitated_guilds_mod
			clr_country_flag = agitated_guilds
			remove_all_rebels_in_country = yes
			add_trust = {
				who = PREV
				value = -100
			}
			add_trust = {
				who = PREV
				value = 50
			}
		}
		reverse_add_aggressive_expansion = {
			who = FROM
			value = -1000
			apply_calc = no
		}
		reverse_remove_opinion = {
			who = FROM
			modifier = declared_war_on_us
		}
		reverse_remove_opinion = {
			who = FROM
			modifier = had_war_with_us
		}
		reverse_remove_opinion = {
			who = FROM
			modifier = caught_spy
		}
		reverse_add_opinion = {
			who = FROM
			modifier = liberated_us_from_nobles
		}
	}
	ai_weight = {
		export_to_variable = {
			variable_name = ai_value
			value = 5000
		}
	}
}
