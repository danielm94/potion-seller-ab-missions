incite_peasants = {
	category = covert
	require_acceptance = no
	is_visible = {
		has_government_mechanic = peasant_zealotry
		NOT = {
			FROM = {
				is_subject_of = ROOT
			}
		}
		FROM = {
			NOT = {
				has_government_attribute = is_peasants_republic
			}
		}
	}
	is_allowed = {
		custom_trigger_tooltip = {
			tooltip = incite_peasants_spy_network_size_tt
			can_incite_peasants = {
				who = FROM
			}
		}
		if = {
			limit = {
				FROM = {
					is_part_of_hre = yes
					has_global_flag = great_peasants_war_flag
				}
			}
			FROM = {
				OR = {
					custom_trigger_tooltip = {
						tooltip = no_radicalized_peasantry_trigger_tt
						any_owned_province = {
							NOT = {
								has_province_modifier = radicalized_peasantry
							}
						}
					}
					custom_trigger_tooltip = {
						tooltip = incite_peasants_on_cooldown_tt
						AND = {
							NOT = {
								has_country_modifier = incited_peasantry
							}
							NOT = {
								has_country_modifier = incited_peasantry_great_peasants_war
							}
						}
					}
				}
			}
		}
		else = {
			custom_trigger_tooltip = {
				tooltip = incite_peasants_on_cooldown_tt
				FROM = {
					NOT = {
						has_country_modifier = incited_peasantry
					}
					NOT = {
						has_country_modifier = incited_peasantry_great_peasants_war
					}
				}
			}
		}
		is_within_range_of_the_peasants_revolution = yes
	}
	on_accept = {
		calculate_spy_action_cost_values = yes
		set_variable = {
			which = spy_network_deduction_var
			which = incite_peasants_cost_var
		}
		multiply_variable = {
			which = spy_network_deduction_var
			value = -1
		}
		variable_change_spy_network_in = {
			who = FROM
			var = spy_network_deduction_var
		}
		reverse_add_opinion = {
			who = FROM
			modifier = incited_peasants_opinion
		}
		FROM = {
			if = {
				limit = {
					is_part_of_hre = yes
					has_global_flag = great_peasants_war_flag
				}
				add_country_modifier = {
					name = incited_peasantry_great_peasants_war
					duration = 3650
				}
			}
			else = {
				add_country_modifier = {
					name = incited_peasantry
					duration = 3650
				}
			}
			if = {
				limit = {
					is_part_of_hre = yes
					has_global_flag = great_peasants_war_flag
				}
				get_number_of_provinces_without_radicalized_peasantry = {
					var_name = num_provinces_without_radicalized_peasantry
				}
				divide_variable = {
					which = num_provinces_without_radicalized_peasantry
					value = 2
				}
				variable_limit_min = {
					var = num_provinces_without_radicalized_peasantry
					min = 1
				}
				for_variable_amount = {
					variable = num_provinces_without_radicalized_peasantry
					effect = "add_radicalized_peasantry_to_random_province_effect = yes"
				}
			}
			else = {
				add_radicalized_peasantry_to_random_province_effect = yes
			}
			add_rebel_progress = {
				rebel_type = anti_tax_rebels
				value = 50
			}
			country_event = {
				id = moldavia_diplomatic_actions_events.1
			}
		}
	}
	ai_will_do = {
		always = no
	}
}

leave_peasant_league = {
	category = league
	require_acceptance = no
	is_visible = {
		is_subject_of = FROM
		is_subject_of_type = allied_commune
	}
	is_allowed = {
		NOT = {
			truce_with = FROM
		}
		OR = {
			NOT = {
				has_opinion = {
					who = FROM
					value = 0
				}
			}
			NOT = {
				trust = {
					who = FROM
					value = 30
				}
			}
			liberty_desire = 50
		}
	}
	on_accept = {
		overlord = {
			add_opinion = {
				who = PREV
				modifier = abandoned_communal_league
			}
			add_casus_belli = {
				type = cb_reintegrate_into_peasant_league
				target = PREV
				months = 300
			}
		}
		grant_independence = yes
		restore_country_color = yes
		FROM = {
			apply_radicalized_peasantry_effect = yes
		}
	}
	ai_will_do = {
		NOT = {
			truce_with = overlord
		}
		OR = {
			NOT = {
				has_opinion = {
					who = overlord
					value = 0
				}
			}
			NOT = {
				trust = {
					who = overlord
					value = 30
				}
			}
			liberty_desire = 50
		}
	}
}

integrate_commune = {
	category = league
	require_acceptance = no
	is_visible = {
		has_government_mechanic = peasant_zealotry
		FROM = {
			OR = {
				is_subject_of_type = allied_commune
				is_subject_of_type = sister_commune
			}
			is_commune_subject_being_integrated = no
		}
	}
	is_allowed = {
		are_we_and_target_at_peace = {
			target = FROM
		}
		FROM = {
			is_commune_loyal_enough_to_integrate = yes
		}
		custom_trigger_tooltip = {
			tooltip = not_currently_integrating_a_commune_subject
			is_integrating_commune_subject = no
		}
		NOT = {
			truce_with = FROM
		}
		NOT = {
			FROM = {
				truce_with = PREV
			}
		}
		custom_trigger_tooltip = {
			tooltip = has_not_integrated_commune_recently_tt
			FROM = {
				OR = {
					NOT = {
						has_country_flag = commune_integrated_recently
					}
					had_country_flag = {
						flag = commune_integrated_recently
						days = 3650
					}
				}
			}
		}
		reverse_has_opinion = {
			who = FROM
			value = 190
		}
		if = {
			limit = {
				FROM = {
					is_subject_of_type = allied_commune
				}
			}
			FROM = {
				trust = {
					who = PREV
					value = 70
				}
			}
		}
		else_if = {
			limit = {
				FROM = {
					is_subject_of_type = sister_commune
				}
			}
			FROM = {
				trust = {
					who = PREV
					value = 100
				}
			}
		}
	}
	on_accept = {
		enable_commune_integration_overlord = yes
		FROM = {
			save_global_event_target_as = commune_being_integrated
			initialize_integration_progress_percentage_variable = yes
			enable_commune_integration_subject = yes
			calculate_monthly_commune_integration_progress = yes
		}
	}
	ai_will_do = {
		always = no
	}
}

export_egyptian_grain = {
	category = economy
	require_acceptance = no
	is_visible = {
		has_country_flag = unlock_egyptian_grain_export
	}
	is_allowed = {
		if = {
			limit = {
				is_rival = FROM
			}
			custom_trigger_tooltip = {
				tooltip = cant_export_grain_to_rival_tt
				always = no
			}
		}
		else_if = {
			limit = {
				is_enemy = FROM
			}
			custom_trigger_tooltip = {
				tooltip = cant_export_grain_to_enemy_tt
				always = no
			}
		}
		else_if = {
			limit = {
				FROM = {
					is_subject_of = root
					any_owned_province = {
						region = egypt_region
						trade_goods = grain
					}
				}
			}
			custom_trigger_tooltip = {
				tooltip = cant_export_grain_to_grain_producer_tt
				always = no
			}
		}
		else = {
			if = {
				limit = {
					NOT = {
						any_owned_province = {
							region = egypt_region
							trade_goods = grain
							base_production = 2
						}
					}
				}
				years_of_income = 0.15
			}
			else = {
				years_of_income = 0.05
			}
			is_at_war = no
			FROM = {
				NOT = {
					has_country_modifier = received_egyptian_grain_shipments
				}
			}
			OR = {
				custom_trigger_tooltip = {
					tooltip = export_egyptian_grain_we_have_grain_tt
					any_owned_province = {
						region = egypt_region
						trade_goods = grain
						base_production = 2
					}
				}
				custom_trigger_tooltip = {
					tooltip = export_egyptian_grain_commune_has_grain_tt
					any_subject_country = {
						is_subject_of_type = integrated_commune
						NOT = {
							liberty_desire = 50
						}
						any_owned_province = {
							region = egypt_region
							trade_goods = grain
							base_production = 2
						}
					}
				}
			}
		}
	}
	on_accept = {
		export_egyptian_grain_cost_effect = yes
		FROM = {
			add_opinion = {
				who = root
				modifier = sent_grain_shipments
			}
			add_trust = {
				who = root
				value = 5
			}
			add_country_modifier = {
				name = received_egyptian_grain_shipments
				duration = 1825
			}
			if = {
				limit = {
					is_subject_of = root
				}
				add_liberty_desire = -10
			}
		}
	}
	ai_will_do = {
		always = no
	}
}
