incite_peasants = {
	category = covert
	require_acceptance = no
	is_visible = {
		has_government_mechanic = peasant_zealotry
		NOT = {
			FROM = {
				is_subject_of = ROOT
			}
		}
		FROM = {
			NOT = {
				has_government_attribute = is_peasants_republic
			}
		}
	}
	is_allowed = {
		custom_trigger_tooltip = {
			tooltip = incite_peasants_spy_network_size_tt
			can_incite_peasants = {
				who = FROM
			}
		}
		custom_trigger_tooltip = {
			tooltip = incite_peasants_on_cooldown_tt
			FROM = {
				OR = {
					NOT = {
						has_country_flag = incited_peasants
					}
					had_country_flag = {
						flag = incited_peasants
						days = 3650
					}
				}
			}
		}
		custom_trigger_tooltip = {
			tooltip = commune_subject_within_border_distance_tt
			OR = {
				NOT = {
					border_distance = {
						who = FROM
						distance = 50
					}
				}
				any_subject_country = {
					is_commune_subject = yes
					NOT = {
						border_distance = {
							who = FROM
							distance = 50
						}
					}
				}
			}
		}
	}
	on_accept = {
		calculate_spy_action_cost_values = yes
		set_variable = {
			which = spy_network_deduction_var
			which = incite_peasants_cost_var
		}
		multiply_variable = {
			which = spy_network_deduction_var
			value = -1
		}
		variable_change_spy_network_in = {
			who = FROM
			var = spy_network_deduction_var
		}
		FROM = {
			if = {
				limit = {
					any_owned_province = {
						culture = bulgarian
						region = balkan_region
					}
				}
				save_global_event_target_as = mop_peasant_subject_tutorial_mission_target
			}
			set_country_flag = incited_peasants
			add_country_modifier = {
				name = incited_peasantry
				duration = 3650
			}
			random_owned_province = {
				add_unrest = 15
			}
			country_event = {
				id = moldavia_diplomatic_actions_events.1
			}
		}
		add_country_modifier = {
			name = incite_peasants_support_buff
			duration = 730
		}
	}
	ai_will_do = {
		always = no
	}
}

	require_acceptance = no
	is_visible = {
		is_subject_of = FROM
		is_subject_of_type = allied_commune
	}
	is_allowed = {
		NOT = {
			truce_with = FROM
		}
		OR = {
			NOT = {
				has_opinion = {
					who = FROM
					value = 0
				}
			}
			NOT = {
				trust = {
					who = FROM
					value = 30
				}
			}
			liberty_desire = 50
		}
	}
	on_accept = {
		overlord = {
			add_opinion = {
				who = PREV
				modifier = abandoned_communal_league
			}
			add_casus_belli = {
				type = cb_reintegrate_into_peasant_league
				target = PREV
				months = 300
			}
		}
		grant_independence = yes
		restore_country_color = yes
	}
	ai_will_do = {
		NOT = {
			truce_with = overlord
		}
		OR = {
			NOT = {
				has_opinion = {
					who = overlord
					value = 0
				}
			}
			NOT = {
				trust = {
					who = overlord
					value = 30
				}
			}
			liberty_desire = 50
		}
	}
}

integrate_commune = {
	category = league
	require_acceptance = no
	is_visible = {
		has_government_mechanic = peasant_zealotry
		FROM = {
			OR = {
				is_subject_of_type = allied_commune
				is_subject_of_type = sister_commune
			}
			is_commune_subject_being_integrated = no
		}
	}
	is_allowed = {
		is_at_war = no
		FROM = {
			is_at_war = no
		}
		FROM = {
			is_commune_loyal_enough_to_integrate = yes
		}
		custom_trigger_tooltip = {
			tooltip = not_currently_integrating_a_commune_subject
			is_integrating_commune_subject = no
		}
		NOT = {
			truce_with = FROM
		}
		NOT = {
			FROM = {
				truce_with = PREV
			}
		}
		custom_trigger_tooltip = {
			tooltip = has_not_integrated_commune_recently_tt
			FROM = {
				OR = {
					NOT = {
						has_country_flag = commune_integrated_recently
					}
					had_country_flag = {
						flag = commune_integrated_recently
						days = 3650
					}
				}
			}
		}
		reverse_has_opinion = {
			who = FROM
			value = 190
		}
		if = {
			limit = {
				FROM = {
					is_subject_of_type = allied_commune
				}
			}
			FROM = {
				trust = {
					who = PREV
					value = 70
				}
			}
		}
		else_if = {
			limit = {
				FROM = {
					is_subject_of_type = sister_commune
				}
			}
			FROM = {
				trust = {
					who = PREV
					value = 100
				}
			}
		}
	}
	on_accept = {
		enable_commune_integration_overlord = yes
		FROM = {
			save_global_event_target_as = commune_being_integrated
			initialize_integration_progress_percentage_variable = yes
			enable_commune_integration_subject = yes
			calculate_monthly_commune_integration_progress = yes
		}
	}
	ai_will_do = {
		always = no
	}
}
