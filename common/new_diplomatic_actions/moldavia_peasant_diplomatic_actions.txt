invite_to_peasant_league = {
	category = league
	alert_index = 3
	alert_tooltip = PEASANT_LEAGUE_INVITE
	require_acceptance = yes
	is_visible = {
		has_government_mechanic = peasant_zealotry
		FROM = {
			has_government_attribute = is_peasants_republic

			who = FROM

		calculate_total_development_of_commune_subjects = yes

			faction = leaguists_faction
			influence = 5
		}
		if = {
			limit = {
				FROM = {
					is_part_of_hre = yes
				}
			}
			FROM = {
				every_owned_province = {
					limit = {
						is_part_of_hre = yes
					}
					set_in_empire = no
					emperor = {
						add_opinion = {
							who = ROOT
							modifier = hre_province_left
						}
					}
				}
			}
			change_variable = {
				hre_nations_diplo_annexed = 1
			}
		}
	}
	on_decline = {
		add_prestige = -5
		add_casus_belli = {
			type = cb_force_into_peasant_league
			target = FROM
			months = 60
		}
		add_opinion = {

			modifier = rejected_alliance
		}
		if = {

				ai = yes
			}

		}

			name = peasant_zeal

				which = ai_value
				which = peasant_zeal_amount
			}
			divide_variable = {
				which = ai_value
				value = 5
			}
		}
		add_entry = {
			name = DIPLO_OPPINION
			export_to_variable = {
				variable_name = ai_value
				value = opinion
				who = FROM
				with = ROOT
			}
			multiply_variable = {
				which = ai_value
				value = 0.125
			}
		}
		add_entry = {
			name = AI_TRUST_TOWARDS_ROOT
			export_to_variable = {
				variable_name = ai_value
				value = trust
				who = FROM
				with = ROOT
			}
			subtract_variable = {
				which = ai_value
				value = 50
			}
			#1 trust = +4 ai points
			multiply_variable = {
				which = ai_value
				value = 0.4
			}

		add_entry = {
			name = AI_DIPLOMATIC_REPUTATION
			export_to_variable = {
				variable_name = ai_value
				value = trigger_value:diplomatic_reputation
			}

			multiply_variable = {
				which = ai_value
				value = 3

leave_peasant_league = {
	category = league
	require_acceptance = no
	is_visible = {
		is_subject_of = FROM
		is_subject_of_type = allied_commune
	}
	is_allowed = {
		NOT = {
			truce_with = FROM
		}
		OR = {
			NOT = {
				has_opinion = {
					who = FROM
					value = 0
				}
			}
			NOT = {
				trust = {
					who = FROM
					value = 30
				}
			}
			liberty_desire = 50
		}
	}
	on_accept = {
		if = {
			limit = {
				has_country_flag = model_commune
			}
			clr_country_flag = model_commune
			remove_opinion = {
				who = overlord
				modifier = designated_us_as_model_comune
			}
		}
		overlord = {
			add_opinion = {
				who = PREV
				modifier = abandoned_communal_league
			}
			add_casus_belli = {
				type = cb_reintegrate_into_peasant_league
				target = PREV
				months = 300
			}
		}
		grant_independence = yes
		restore_country_color = yes
		FROM = {
			apply_radicalized_peasantry_effect = yes
		}
	}
	ai_will_do = {
		NOT = {
			truce_with = overlord
		}
		OR = {
			NOT = {
				has_opinion = {
					who = overlord
					value = 0
				}
			}
			NOT = {
				trust = {
					who = overlord
					value = 30
				}
			}
			liberty_desire = 50
		}
	}
}

enforce_peace_commune = {
	category = league
	require_acceptance = no
	is_visible = {
		has_government_mechanic = peasant_zealotry
		FROM = {
			is_subject_of_type = allied_commune
			is_at_war = yes
		}
	}
	is_allowed = {
		custom_trigger_tooltip = {
			tooltip = is_in_offensive_war_against_commune_subject_tt
			FROM = {
				any_war_enemy_country = {
					is_commune_subject = yes
					is_subject_of = ROOT
					is_in_war = {
						defender_leader = THIS
						attacker_leader = FROM
					}
				}
			}
		}
		FROM = {
			NOT = {
				liberty_desire = 50
			}
		}
		has_government_power = {
			mechanic_type = peasant_zealotry
			power_type = peasant_zeal
			value = 5
		}
	}
	on_accept = {
		change_peasant_zeal = {
			value = -5
			scaled_with_gain_modifier = no
		}
		FROM = {
			every_war_enemy_country = {
				limit = {
					is_in_war = {
						attacker_leader = FROM
						defender_leader = this
					}
				}
				PREV = {
					white_peace = PREV
				}
			}
			add_liberty_desire = 10
			set_country_flag = cant_declare_war_on_communes_10_years
		}
	}
	ai_will_do = {
		always = no
	}
}

designate_model_commune = {
	category = league
	require_acceptance = no
	is_visible = {
		has_government_mechanic = peasant_zealotry
		FROM = {
			is_subject_of_type = allied_commune
			NOT = {
				has_country_flag = model_commune
			}
		}
	}
	is_allowed = {
		if = {
			limit = {
				OR = {
					NOT = {
						has_country_flag = set_model_commune
					}
					had_country_flag = {
						flag = set_model_commune
						days = 3650
					}
				}
			}
			has_government_power = {
				mechanic_type = peasant_zealotry
				power_type = peasant_zeal
				value = 10
			}
		}
		else = {
			custom_trigger_tooltip = {
				tooltip = has_designated_model_commune_recently_tt
				always = no
			}
		}
	}
	on_accept = {
		change_peasant_zeal = {
			value = -10
			scaled_with_gain_modifier = no
		}
		if = {
			limit = {
				any_country = {
					has_country_flag = model_commune
				}
			}
			random_country = {
				limit = {
					has_country_flag = model_commune
				}
				clr_country_flag = model_commune
				remove_opinion = {
					who = root
					modifier = designated_us_as_model_comune
				}
			}
		}
		set_country_flag = set_model_commune
		add_country_modifier = {
			name = designated_model_commune
			duration = -1
		}
		reverse_add_opinion = {
			who = from
			modifier = designated_us_as_model_comune
		}
		from = {
			set_country_flag = model_commune
			save_global_event_target_as = model_commune_country
			add_liberty_desire = -10
		}
	}
	ai_will_do = {
		always = no
	}
}

						}
					}
				}
				years_of_income = 0.15
			}
			else = {
				years_of_income = 0.05
			}

			}
