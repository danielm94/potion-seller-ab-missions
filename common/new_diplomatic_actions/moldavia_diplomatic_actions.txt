incite_peasants = {
	category = covert
	require_acceptance = no
	is_visible = {
		has_government_mechanic = peasant_zealotry
		NOT = {
			FROM = {
				is_subject_of = ROOT
			}
		}
		FROM = {
			is_a_peasant_government = no
		}
	}
	is_allowed = {
		custom_trigger_tooltip = {
			tooltip = incite_peasants_spy_network_size_tt
			can_incite_rebels = {
				who = FROM
			}
		}
		custom_trigger_tooltip = {
			tooltip = incite_peasants_on_cooldown_tt
			FROM = {
				OR = {
					NOT = {
						has_country_flag = incited_peasants
					}
					had_country_flag = {
						flag = incited_peasants
						days = 3650
					}
				}
			}
		}
	}
	on_accept = {
		calculate_spy_action_cost_values = yes
		set_variable = {
			which = spy_network_deduction_var
			which = incite_rebels_cost_var
		}
		multiply_variable = {
			which = spy_network_deduction_var
			value = -1
		}
		variable_change_spy_network_in = {
			who = FROM
			var = spy_network_deduction_var
		}
		FROM = {
			set_country_flag = incited_peasants
			add_country_modifier = {
				name = incited_peasantry
				duration = 3650
			}
			random_owned_province = {
				add_unrest = 15
			}
			country_event = {
				id = moldavia_diplomatic_actions_events.1
			}
		}
	}
	ai_will_do = {
		always = no
	}
}

integrate_commune = {
	category = influence
	require_acceptance = no
	is_visible = {
		has_government_mechanic = peasant_zealotry
		FROM = {
			OR = {
				is_subject_of_type = allied_commune
				is_subject_of_type = sister_commune
			}
			is_commune_subject_being_integrated = no
		}
	}
	is_allowed = {
		is_at_war = no
		FROM = {
			is_at_war = no
		}
		FROM = {
			is_commune_loyal_enough_to_integrate = yes
		}
		custom_trigger_tooltip = {
			tooltip = not_currently_integrating_a_commune_subject
			is_integrating_commune_subject = no
		}
		NOT = {
			truce_with = FROM
		}
		NOT = {
			FROM = {
				truce_with = PREV
			}
		}
		custom_trigger_tooltip = {
			tooltip = has_not_integrated_commune_recently_tt
			FROM = {
				OR = {
					NOT = {
						has_country_flag = commune_integrated_recently
					}
					had_country_flag = {
						flag = commune_integrated_recently
						days = 3650
					}
				}
			}
		}
		reverse_has_opinion = {
			who = FROM
			value = 190
		}
		if = {
			limit = {
				FROM = {
					is_subject_of_type = allied_commune
				}
			}
			FROM = {
				trust = {
					who = PREV
					value = 70
				}
			}
		}
		else_if = {
			limit = {
				FROM = {
					is_subject_of_type = sister_commune
				}
			}
			FROM = {
				trust = {
					who = PREV
					value = 100
				}
			}
		}
	}
	on_accept = {
		enable_commune_integration_overlord = yes
		FROM = {
			save_global_event_target_as = commune_being_integrated
			initialize_integration_progress_percentage_variable = yes
			enable_commune_integration_subject = yes
			calculate_monthly_commune_integration_progress = yes
		}
	}
	ai_will_do = {
		always = no
	}
}

		}
		FROM = {
			calculate_monthly_commune_integration_progress = yes
		}
		calculate_overlord_adm_investment_integration_progress_gain = yes
	}
	ai_will_do = {
		always = no

stage_coup_commune_integration = {
	category = covert
	require_acceptance = no
	is_visible = {
		has_government_mechanic = peasant_zealotry
		FROM = {
			is_commune_subject_being_integrated = yes
			is_subject_of = ROOT
		}
	}
	is_allowed = {
		custom_trigger_tooltip = {
			tooltip = stage_coup_spy_network_size_tt
			can_stage_coup = {
				who = FROM
			}
		}
		variable_arithmetic_trigger = {
			custom_tooltip = has_enough_ducats_for_coup_tt
			export_to_variable = {
				which = target_dev
				value = total_development
				who = FROM
			}
			export_to_variable = {
				which = ducats
				value = treasury
			}
			check_variable = {
				which = ducats
				which = target_dev
			}
		}
		custom_trigger_tooltip = {
			tooltip = coup_leader_still_alive_tt
			FROM = {
				NOT = {
					has_ruler_flag = commune_coup_leader
				}
			}
		}
	}
	on_accept = {
		calculate_spy_action_cost_values = yes
		set_variable = {
			which = spy_network_deduction_var
			which = stage_coup_cost_var
		}
		multiply_variable = {
			which = spy_network_deduction_var
			value = -1
		}
		variable_change_spy_network_in = {
			who = FROM
			var = spy_network_deduction_var
		}
		export_to_variable = {
			which = total_dev
			value = total_development
			who = FROM
		}
		multiply_variable = {
			total_dev = -1
		}
		variable_effect = {
			var = total_dev
			effect = add_treasury
		}
		FROM = {
			kill_ruler = yes
			set_ruler_flag = commune_coup_leader
			change_commune_integration_progress_as_percentage_of_cost = {
				percentage = 0.2
			}
			add_liberty_desire = 40
			add_opinion = {
				who = overlord
				modifier = commune_staged_coup_against_us
			}
			add_trust = {
				who = overlord
				value = -40
			}
		}
		every_subject_country = {
			limit = {
				NOT = {
					tag = FROM
				}
			}
			add_liberty_desire = 20
			add_trust = {
				who = overlord
				value = -20
			}
			add_opinion = {
				who = overlord
				modifier = commune_staged_coup_against_another_commune
			}
		}
	}
	ai_will_do = {
		always = no
	}
}
