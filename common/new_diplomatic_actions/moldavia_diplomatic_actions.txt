incite_peasants = {
	category = covert
	require_acceptance = no
	# Determines when the action is visible
	is_visible = {
		has_government_attribute = can_incite_peasants
		FROM = {
			is_a_peasant_government = no
		}
	}
	is_allowed = {
		custom_trigger_tooltip = {
			tooltip = incite_peasants_spy_network_size_tt
			can_incite_rebels = {
				who = FROM
			}
		}
		custom_trigger_tooltip = {
			tooltip = incite_peasants_on_cooldown_tt
			FROM = {
				OR = {
					NOT = {
						has_country_flag = incited_peasants
					}
					had_country_flag = {
						flag = incited_peasants
						days = 3650
					}
				}
			}
		}
	}
	on_accept = {
		calculate_incite_rebels_cost = yes
		set_variable = {
			which = spy_network_deduction_var
			which = incite_rebels_cost_var
		}
		multiply_variable = {
			which = spy_network_deduction_var
			value = -1
		}
		variable_change_spy_network_in = {
			who = FROM
			var = spy_network_deduction_var
		}
		FROM = {
			set_country_flag = incited_peasants
			add_country_modifier = {
				name = incited_peasantry
				duration = 3650
			}
			random_owned_province = {
				add_unrest = 15
			}
			country_event = {
				id = moldavia_diplomatic_actions_events.1
			}
		}
	}
	ai_will_do = {
		always = no
	}
}

integrate_commune = {
	category = influence
	require_acceptance = no
	# Determines when the action is visible
	is_visible = {
		has_government_mechanic = peasant_zealotry
		FROM = {
			OR = {
				is_subject_of_type = allied_commune
				is_subject_of_type = sister_commune
			}
			is_commune_subject_being_integrated = no
		}
	}
	is_allowed = {
		is_at_war = no
		FROM = {
			is_at_war = no
		}
		FROM = {
			is_commune_loyal_enough_to_integrate = yes
		}
		custom_trigger_tooltip = {
			tooltip = not_currently_integrating_a_commune_subject
			is_integrating_commune_subject = no
		}
		NOT = {
			truce_with = FROM
		}
		NOT = {
			FROM = {
				truce_with = PREV
			}
		}
		custom_trigger_tooltip = {
			tooltip = has_not_integrated_commune_recently_tt
			FROM = {
				OR = {
					NOT = {
						has_country_flag = commune_integrated_recently
					}
					had_country_flag = {
						flag = commune_integrated_recently
						days = 3650
					}
				}
			}
		}
		reverse_has_opinion = {
			who = FROM
			value = 190
		}
		if = {
			limit = {
				FROM = {
					is_subject_of_type = allied_commune
				}
			}
			FROM = {
				trust = {
					who = PREV
					value = 70
				}
			}
		}
		else_if = {
			limit = {
				FROM = {
					is_subject_of_type = sister_commune
				}
			}
			FROM = {
				trust = {
					who = PREV
					value = 100
				}
			}
		}
	}
	on_accept = {
		enable_commune_integration_overlord = yes
		FROM = {
			save_global_event_target_as = commune_being_integrated
			initialize_integration_progress_percentage_variable = yes
			enable_commune_integration_subject = yes
			calculate_monthly_commune_integration_progress = yes
			calculate_overlord_adm_investment_integration_progress_gain = yes
			calculate_overlord_dip_investment_integration_progress_gain = yes
		}
	}
	ai_will_do = {
		always = no
	}
}

cancel_commune_integration = {
	category = influence
	require_acceptance = no
	# Determines when the action is visible
	is_visible = {
		has_government_mechanic = peasant_zealotry
		FROM = {
			is_commune_subject_being_integrated = yes
		}
	}
	is_allowed = {
		always = yes
	}
	on_accept = {
		disable_commune_integration_overlord = yes
		FROM = {
			disable_commune_integration_subject = yes
			initialize_commune_integration_progress = yes
			initialize_integration_progress_percentage_variable = yes
		}
		clear_global_event_target = commune_being_integrated
		clear_commune_integration_investments = yes
	}
	ai_will_do = {
		always = no
	}
}

invest_adm_commune_integration = {
	category = influence
	require_acceptance = no
	is_visible = {
		is_investing_adm_power_in_commune = no
		has_government_mechanic = peasant_zealotry
		FROM = {
			is_commune_subject_being_integrated = yes
		}
	}
	is_allowed = {
		custom_trigger_tooltip = {
			tooltip = not_investing_mana_in_commune_tt
			is_investing_mana_in_commune = no
		}
		adm_power = 1
		variable_arithmetic_trigger = {
			custom_tooltip = adm_efficiency_at_least_40_tt
			export_to_variable = {
				which = admin_eff
				value = modifier:administrative_efficiency
			}
			check_variable = {
				admin_eff = -0.39
			}
		}
		num_of_diplomats = 1
		FROM = {
			is_commune_loyal_enough_to_integrate = yes
		}
	}
	on_accept = {
		set_country_flag = investing_adm_in_commune
		add_country_modifier = {
			name = investing_admin_commune
			duration = -1
		}
		FROM = {
			calculate_monthly_commune_integration_progress = yes
		}
		calculate_overlord_adm_investment_integration_progress_gain = yes
		calculate_overlord_dip_investment_integration_progress_gain = yes
	}
	ai_will_do = {
		always = no
	}
}
