# condition				A diplomatic action can have any number of condition blocks, each with its own
#						tooltip, potential and allow section
#
# 	tooltip					Sets a custom text string similar to the hardcoded limits
# 							If no tooltip is scripted, the tooltip for the actual trigger will be shown
#							Note that the custom tooltip is only shown if the allow trigger is NOT met
#
# 	potential				Determines if the trigger is applicable or not
# 	allow					Determines if the action is valid or not
#
# effect				A diplomatic action can only have one effect block
# ROOT					actor
# FROM					target

royal_marriage = {
	condition = {
		tooltip = MARWAR
		potential = {
		}
		allow = {
			NOT = {
				war_with = FROM
			}
		}
	}
	condition = {
		tooltip = tuh_from_cant_marry_tooltip
		potential = {
			FROM = {
				has_government_attribute = can_only_marry_subjects
			}
			NOT = {
				is_subject_of = FROM
			}
		}
		allow = {
			always = no
		}
	}
	condition = {
		tooltip = tuh_root_cant_marry_tooltip
		potential = {
			has_government_attribute = can_only_marry_subjects
			NOT = {
				FROM = {
					is_subject_of = ROOT
				}
			}
		}
		allow = {
			always = no
		}
	}
	effect = {
	}
}

form_coalition = {
	condition = {
		tooltip = form_coalition_tooltip
		potential = {
		}
		allow = {
			has_opinion_modifier = {
				modifier = aggressive_expansion
				who = FROM
			}
			if = {
				limit = {
					FROM = {
						has_country_flag = wal_increased_coalition_threshold
						has_country_flag = wal_the_final_lesson_coalition_threshold
					}
				}
				NOT = {
					has_opinion_modifier = {
						modifier = aggressive_expansion
						who = FROM
						value = -99
					}
				}
			}
			else_if = {
				limit = {
					FROM = {
						OR = {
							has_country_flag = wal_increased_coalition_threshold
							has_country_flag = wal_the_final_lesson_coalition_threshold
						}
					}
				}
				NOT = {
					has_opinion_modifier = {
						modifier = aggressive_expansion
						who = FROM
						value = -74
					}
				}
			}
			else = {
				NOT = {
					has_opinion_modifier = {
						modifier = aggressive_expansion
						who = FROM
						value = -49
					}
				}
			}
		}
	}
}

support_rebels = {
	pre_effect = {
		if = {
			limit = {
				has_country_flag = defenestration_tactics_enabled
				NOT = {
					has_country_modifier = mop_successful_defenestration
				}
			}
			random = {
				chance = 10
				set_country_flag = defenestration_happened
				country_event = {
					id = moldavia_peasant_events.10
				}
			}
		}
		if = {
			limit = {
				has_country_flag = come_out_ye_black_and_tans
				NOT = {
					has_country_flag = defenestration_happened
				}
				NOT = {
					has_country_modifier = mop_successful_bombing
				}
			}
			random = {
				chance = 10
				country_event = {
					id = moldavia_peasant_events.12
				}
			}
		}
		clr_country_flag = defenestration_happened
		FROM = {
			set_country_flag = support_rebels_against_@THIS
			PREV = {
				set_country_flag = support_rebels_against_@PREV
			}
			every_owned_province = {
				limit = {
					has_radicalized_peasantry_province = yes
					NOT = {
						has_province_modifier = recent_uprising
					}
				}
				set_province_unrest_to_value_if_negative = {
					value = 10
				}
			}
			country_event = {
				id = moldavia_diplomatic_actions_events.3
				days = 1825
			}
			if = {
				limit = {
					ROOT = {
						OR = {
							NOT = {
								mission_completed = mop_arm_the_bulgar_peasants
							}
							NOT = {
								mission_completed = mop_war_for_the_bulgars
							}
							NOT = {
								mission_completed = mop_liberate_the_bulgars
							}
							NOT = {
								mission_completed = mop_bulgars_are_free
							}
						}
					}
					any_owned_province = {
						culture = bulgarian
						region = balkan_region
					}
					has_radicalized_peasantry_country = yes
				}
				save_global_event_target_as = mop_peasant_subject_tutorial_mission_target
			}
		}
	}
}
