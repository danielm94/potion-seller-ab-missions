decimal_decomposition_integers_effect = {

				var = $var$
				value = 0
			}
		}
		set_variable = {
			which = modulo
			which = $var$
		}
		modulo_variable = {
			which = modulo
			value = 10
		}
		divide_variable = {
			which = $var$
			value = 10
		}
		#-1 rounds down, effectively truncating the decimal
		round_variable = {
			which = $var$
			value = -1
		}
		while = {
			limit = {
				check_variable = {
					which = modulo
					value = 1
				}
			}
			if = {
				limit = {
					check_variable = {
						which = is_negative
						value = 1
					}
				}
				$effect$ = -$increment$
			}
			else = {
				$effect$ = $increment$
			}
			subtract_variable = {
				which = modulo
				value = 1
			}
		}
	}
}

variable_change_spy_network_in = {
	set_variable = {
		which = num
		which = $var$
	}
	#Truncate any decimals
	round_variable = {
		which = num
		value = -1
	}
	set_variable = {
		which = is_negative
		value = 0
	}
	if = {
		limit = {
			NOT = {
				check_variable = {
					which = num
					value = 0
				}
			}
		}
		set_variable = {
			which = is_negative
			value = 1
		}
		multiply_variable = {
			which = num
			value = -1
		}
	}
	change_spy_network_in_effect = {
		var = num
		who = $who$
		increment = 1
	}
	change_spy_network_in_effect = {
		var = num
		who = $who$
		increment = 10
	}
	change_spy_network_in_effect = {
		var = num
		who = $who$
		increment = 100
	}
}

change_spy_network_in_effect = {
	if = {
		limit = {
			is_variable_greater_than = {
				var = $var$
				value = 0
			}
		}
		set_variable = {
			which = modulo
			which = $var$
		}
		modulo_variable = {
			which = modulo
			value = 10
		}
		divide_variable = {
			which = $var$
			value = 10
		}
		#-1 rounds down, effectively truncating the decimal
		round_variable = {
			which = $var$
			value = -1
		}
		while = {
			limit = {
				check_variable = {
					which = modulo
					value = 1
				}
			}
			if = {
				limit = {
					check_variable = {
						which = is_negative
						value = 1
					}
				}
				add_spy_network_in = {
					who = $who$
					value = -$increment$
				}
			}
			else = {
				add_spy_network_in = {
					who = $who$
					value = $increment$
				}
			}
			subtract_variable = {
				which = modulo
				value = 1
			}
		}
	}
}

variable_truncate_decimal = {
	if = {
		limit = {
			check_variable = {
				$var$ = 0
			}
		}
		round_variable = {
			which = $var$
			value = -1
		}
	}
	else = {
		round_variable = {
			which = $var$
			value = 1
		}
	}
}

variable_make_positive = {
	if = {
		limit = {
			NOT = {
				check_variable = {
					$var$ = 0
				}
			}
		}
		multiply_variable = {
			$var$ = -1
		}
	}
}
