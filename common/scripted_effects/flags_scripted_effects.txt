toggle_country_flag = {
	if = {
		limit = {
			has_country_flag = $flag$
		}
		clr_country_flag = $flag$
	}
	else = {
		set_country_flag = $flag$
	}
}

toggle_country_flag_with_sound = {
	if = {
		limit = {
			has_country_flag = $flag$
		}
		clr_country_flag = $flag$
		play_sound = lock_mapmode
	}
	else = {
		set_country_flag = $flag$
		play_sound = unlock_mapmode
	}
}
