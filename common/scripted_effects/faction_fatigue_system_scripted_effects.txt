#SCOPE = COUNTRY
faction_fatigue_initialize_faction_fatigue_variables = {
	set_variable = {
		new_exemplarists_count = 0
	}
	set_variable = {
		new_leaguists_count = 0
	}
	set_variable = {
		new_abolitionists_count = 0
	}
	set_variable = {
		fatigue_gain_modifier_min = 1
	}
	set_variable = {
		fatigue_gain_modifier_max = 2
	}
	set_variable = {
		fatigue_gain_modifier_new_candidate_change = 0.1
	}
	set_variable = {
		fatigue_gain_modifier_reelect_candidate_change = -0.1
	}
	set_variable = {
		fatigue_gain_modifier_parliament_on_action_change = -0.05
	}
	set_variable = {
		fatigue_gain_modifier_parliament_issue_change = -0.2
	}
	set_variable = {
		min_faction_fatigue_new_candidate_change = 0.15
	}
	set_variable = {
		min_faction_fatigue_reelect_candidate_change = -0.25
	}
	set_variable = {
		min_faction_fatigue_parliament_on_action_change = -0.125
	}
	set_variable = {
		min_faction_fatigue_parliament_issue_change = -0.5
	}
	set_variable = {
		which = current_fatigue_gain_modifier
		which = fatigue_gain_modifier_min
	}
	set_variable = {
		faction_fatigue_min = 0
	}
	set_variable = {
		faction_fatigue_max = 10
	}
	set_variable = {
		base_corruption_penalty = 0.5
	}
	set_variable = {
		base_stability_penalty = -0.6
	}
	set_variable = {
		base_unrest_penalty = 3
	}
	set_variable = {
		base_peasant_zeal_penalty = -1.5
	}
}

faction_fatigue_change_fatigue_gain_modifier = {
	change_variable = {
		which = current_fatigue_gain_modifier
		which = $var$
	}
	variable_clamp = {
		var = current_fatigue_gain_modifier
		var_min = fatigue_gain_modifier_min
		var_max = fatigue_gain_modifier_max
	}
}

	change_variable = {
		which = faction_fatigue_min
		which = $var$
	}
	variable_limit_min = {
		var = faction_fatigue_min
		min = 0
	}
}

#SCOPE = COUNTRY
faction_fatigue_calculate_new_candidate_equilibrium = {
	set_variable = {
		which = $var_name$
		which = new_exemplarists_count
	}
	change_variable = {
		which = $var_name$
		which = new_leaguists_count
	}
	change_variable = {
		which = $var_name$
		which = new_abolitionists_count
	}
	divide_variable = {
		$var_name$ = 3
	}
}

# SCOPE = COUNTRY
# Excess = faction new candidates - (total number of new candidates / 3)
# If excess < 0, excess = 0
faction_fatigue_calculate_faction_excess = {
	faction_fatigue_calculate_new_candidate_equilibrium = {
		var_name = new_candidate_equilibrium
	}
	set_variable = {
		which = $var_name$
		which = new_$faction$_count
	}
	subtract_variable = {
		which = $var_name$
		which = new_candidate_equilibrium
	}
	variable_limit_min = {
		which = $var_name$
		min = 0
	}
}

#SCOPE = COUNTRY
faction_fatigue_calculate_faction_fatigue = {
	faction_fatigue_calculate_faction_excess = {
		var_name = $faction$_excess
		faction = $faction$
	}
	set_variable = {
		which = $var_name$
		which = $faction$_excess
	}
	multiply_variable = {
		which = $var_name$
		which = current_fatigue_gain_modifier
	}
	variable_clamp = {
		var = $var_name$
		var_min = faction_fatigue_min
		var_max = faction_fatigue_max
	}
}

faction_fatigue_apply_candidate_penalties = {
	if = {
		limit = {
			is_variable_greater_than = {
				var = $faction$_candidate_corruption_penalty
				value = 0
			}
		}
		custom_tooltip = $faction$_candidate_corruption_penalty_tt
		hidden_effect = {
			variable_effect = {
				var = $faction$_candidate_corruption_penalty
				effect = add_corruption
			}
		}
	}
	if = {
		limit = {
			NOT = {
				check_variable = {
					which = $faction$_candidate_stability_penalty
					value = 0
				}
			}
		}
		custom_tooltip = $faction$_candidate_stability_penalty_tt
		hidden_effect = {
			variable_effect = {
				var = $faction$_candidate_stability_penalty
				effect = add_stability
			}
		}
	}
	if = {
		limit = {
			is_variable_greater_than = {
				var = $faction$_candidate_unrest_penalty
				value = 0
			}
		}
		custom_tooltip = $faction$_candidate_unrest_penalty_tt
		hidden_effect = {
			every_owned_province = {
				set_variable = {
					which = $faction$_candidate_unrest_penalty
					which = PREV
				}
				variable_effect = {
					var = $faction$_candidate_unrest_penalty
					effect = add_unrest
				}
			}
		}
	}
	if = {
		limit = {
			NOT = {
				check_variable = {
					which = $faction$_candidate_peasant_zeal_penalty
					value = 0
				}
			}
		}
		custom_tooltip = $faction$_candidate_peasant_zeal_penalty_tt
		hidden_effect = {
			variable_government_power_effect = {
				var = $faction$_candidate_peasant_zeal_penalty
				mechanic_type = peasant_zealotry
				power_type = peasant_zeal
				scaled_with_gain_modifier = no
			}
		}
	}
}
