#SCOPE = COUNTRY
calculate_protector_little_folk_personality_chances_effect = {
	hidden_effect = {
		set_variable = {
			num_allied_communes = 0
		}
		set_variable = {
			num_sister_communes = 0
		}
		set_variable = {
			num_integrated_communes = 0
		}
		every_subject_country = {
			if = {
				limit = {
					is_subject_of_type = allied_commune
				}
				overlord = {
					change_variable = {
						num_allied_communes = 1
					}
				}
			}
			else_if = {
				limit = {
					is_subject_of_type = sister_commune
				}
				overlord = {
					change_variable = {
						num_sister_communes = 1
					}
				}
			}
			else_if = {
				limit = {
					is_subject_of_type = integrated_commune
				}
				overlord = {
					change_variable = {
						num_integrated_communes = 1
					}
				}
			}
		}
		set_variable = {
			which = chance_gain_from_allied_communes
			which = num_allied_communes
		}
		multiply_variable = {
			chance_gain_from_allied_communes = 0.5
		}
		set_variable = {
			which = total_chance_of_little_folk_personality
			which = chance_gain_from_allied_communes
		}
		set_variable = {
			which = chance_gain_from_sister_communes
			which = num_sister_communes
		}
		multiply_variable = {
			chance_gain_from_sister_communes = 0.75
		}
		change_variable = {
			which = total_chance_of_little_folk_personality
			which = chance_gain_from_sister_communes
		}
		set_variable = {
			which = chance_gain_from_integrated_communes
			which = num_integrated_communes
		}
		multiply_variable = {
			chance_gain_from_integrated_communes = 1
		}
		change_variable = {
			which = total_chance_of_little_folk_personality
			which = chance_gain_from_integrated_communes
		}
		if = {
			limit = {
				allied_commune = 5
				check_variable = {
					total_chance_of_little_folk_personality = 100
				}
			}
			set_variable = {
				total_chance_of_little_folk_personality = 100
			}
		}
	}
}

#SCOPE = COUNTRY
add_protector_little_folk_personality_on_election_effect = {
	custom_tooltip = chance_of_getting_protector_little_folk_personality_tt
	display_verbose_election_event_tooltip_if_toggled = {
		tooltip = chance_of_getting_protector_little_folk_personality_math_tt
	}
	if = {
		limit = {
			NOT = {
				ruler_has_personality = peasant_revolt_leader_personality
			}
		}
		hidden_effect = {
			random_variable = {
				which = personality_roll
				value = 100
			}
			if = {
				limit = {
					OR = {
						NOT = {
							check_variable = {
								which = personality_roll
								which = total_chance_of_little_folk_personality
							}
						}
						is_variable_equal = {
							which = personality_roll
							which = total_chance_of_little_folk_personality
						}
					}
				}
				get_number_of_ruler_personalities = {
					var_name = num_personalities
				}
				if = {
					limit = {
						check_variable = {
							num_personalities = 3
						}
					}
					country_event = {
						id = peasant_election_events.2
					}
				}
				else = {
					add_ruler_personality = peasant_revolt_leader_personality
				}
			}
		}
	}
}
