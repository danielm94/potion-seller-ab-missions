#SCOPE = COUNTRY
change_razesi_power = {
	add_government_power = {
		mechanic_type = boieri_vs_razesi_mechanic
		power_type = boieri_vs_razesi_power
		value = $value$
	}
}

#SCOPE = COUNTRY
boieri_vs_razesi_on_development_effect = {
	if = {
		limit = {
			owner = {
				has_government_mechanic = boieri_vs_razesi_mechanic
			}
		}
		if = {
			limit = {
				is_province_a_boieri_province = yes
			}
			owner = {
				change_razesi_power = {
					value = -0.1
				}
			}
		}
		else_if = {
			limit = {
				is_province_a_razesi_province = yes
			}
			owner = {
				change_razesi_power = {
					value = 0.1
				}
			}
		}
	}
}

#SCOPE = PROVINCE
on_province_owner_change_conquest_mol = {
	if = {
		limit = {
			MOL = {
				has_government_mechanic = boieri_vs_razesi_mechanic
			}
		}
		export_to_variable = {
			which = province_dev_var
			value = development
		}
		multiply_variable = {
			which = province_dev_var
			value = 0.25
		}
		if = {
			limit = {
				is_province_a_boieri_province = yes
			}
			multiply_variable = {
				which = province_dev_var
				value = -1
			}
			MOL = {
				set_variable = {
					which = province_dev_var
					which = PREV
				}
				variable_government_power_effect = {
					var = province_dev_var
					mechanic_type = boieri_vs_razesi_mechanic
					power_type = boieri_vs_razesi_power
					scaled_with_gain_modifier = no
				}
			}
		}
		else_if = {
			limit = {
				is_province_a_razesi_province = yes
			}
			MOL = {
				set_variable = {
					which = province_dev_var
					which = PREV
				}
				variable_government_power_effect = {
					var = province_dev_var
					mechanic_type = boieri_vs_razesi_mechanic
					power_type = boieri_vs_razesi_power
					scaled_with_gain_modifier = no
				}
			}
		}
	}
}
