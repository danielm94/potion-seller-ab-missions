apply_radicalized_peasantry_effect = {
	clear_all_radicalized_peasantry = yes
	apply_radicalized_peasantry_to_all_provinces = yes
	clear_neighboring_peasant_opinion = yes
	apply_neighboring_peasant_opinion = yes
}

#SCOPE = COUNTRY
clear_all_radicalized_peasantry = {
	every_province = {
		limit = {
			has_province_modifier = radicalized_peasantry
			owner = {
				NOT = {
					has_country_modifier = incited_peasantry
				}
				NOT = {
					has_country_modifier = incited_peasantry_great_peasants_war
				}
			}
		}
		remove_province_modifier = radicalized_peasantry
	}
}

#SCOPE = COUNTRY
apply_radicalized_peasantry_to_all_provinces = {
	every_country = {
		limit = {
			is_country_that_spreads_radicalized_peasants = yes
		}
		apply_radicalized_peasantry_to_neighbors = yes
	}
}

#SCOPE = COUNTRY
apply_radicalized_peasantry_to_neighbors = {
	every_neighbor_province = {
		limit = {
			owner = {
				NOT = {
					has_government_attribute = is_peasants_republic
				}
				has_nobility_estate_of_some_kind = yes
				NOT = {
					is_subject_of_type = consulate_subject
				}
			}
		}
		add_province_modifier = {
			name = radicalized_peasantry
			duration = -1
		}
	}
}

clear_neighboring_peasant_opinion = {
	every_country = {
		limit = {
			has_been_spreader_of_radicalized_peasants = yes
		}
		every_known_country = {
			limit = {
				has_opinion_modifier = {
					who = PREV
					modifier = peasant_scum_opinion
				}
			}
			remove_opinion = {
				who = PREV
				modifier = peasant_scum_opinion
			}
			reverse_remove_opinion = {
				who = PREV
				modifier = tyrant_scum_opinion
			}
		}
	}
}

apply_neighboring_peasant_opinion = {
	every_country = {
		limit = {
			is_country_that_spreads_radicalized_peasants = yes
		}
		every_neighbor_country = {
			limit = {
				NOT = {
					has_government_attribute = is_peasants_republic
				}
				has_nobility_estate_of_some_kind = yes
				NOT = {
					is_subject_of_type = consulate_subject
				}
			}
			add_opinion = {
				who = PREV
				modifier = peasant_scum_opinion
			}
			reverse_add_opinion = {
				who = PREV
				modifier = tyrant_scum_opinion
			}
			if = {
				limit = {
					PREV = {
						overlord = {
							is_country_that_spreads_radicalized_peasants = yes
						}
					}
				}
				save_event_target_as = afflicted_by_peasants_country
				PREV = {
					overlord = {
						add_opinion = {
							who = event_target:afflicted_by_peasants_country
							modifier = tyrant_scum_opinion
						}
						reverse_add_opinion = {
							who = event_target:afflicted_by_peasants_country
							modifier = peasant_scum_opinion
						}
					}
				}
			}
		}
	}
}

#SCOPE = COUNTRY
get_number_of_provinces_without_radicalized_peasantry = {
	export_to_variable = {
		which = state_provinces
		value = trigger_value:num_of_provinces_in_states
	}
	export_to_variable = {
		which = territory_provinces
		value = trigger_value:num_of_provinces_in_territories
	}
	set_variable = {
		which = number_of_provinces
		which = state_provinces
	}
	change_variable = {
		which = number_of_provinces
		which = territory_provinces
	}
	set_variable = {
		num_of_radicalized_peasantry = 0
	}
	every_owned_province = {
		limit = {
			has_province_modifier = radicalized_peasantry
		}
		owner = {
			change_variable = {
				num_of_radicalized_peasantry = 1
			}
		}
	}
	set_variable = {
		which = $var_name$
		which = number_of_provinces
	}
	subtract_variable = {
		which = $var_name$
		which = num_of_radicalized_peasantry
	}
}

#SCOPE = COUNTRY
add_radicalized_peasantry_to_random_province_effect = {
	random_owned_province = {
		limit = {
			NOT = {
				has_province_modifier = radicalized_peasantry
			}
		}
		add_province_modifier = {
			name = radicalized_peasantry
			duration = 3650
		}
		save_event_target_as = radicalized_province
	}
}
