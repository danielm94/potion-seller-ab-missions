change_peasant_zeal = {
	add_government_power = {
		mechanic_type = peasant_zealotry
		power_type = peasant_zeal
		value = $value$
		[[scaled_with_gain_modifier]
		scaled_with_gain_modifier = $scaled_with_gain_modifier$
		]
	}
}

#SCOPE = COUNTRY
calculate_spy_action_cost_values = {
	calculate_incite_peasants_cost = yes
	calculate_stage_coup_cost = yes
}

#SCOPE = COUNTRY
get_incite_peasants_base_cost_var = {
	set_variable = {
		$var$ = 30
	}
}

#SCOPE = COUNTRY
get_stage_coup_base_cost_var = {
	set_variable = {
		$var$ = 70
	}
}

#SCOPE = COUNTRY
calculate_incite_peasants_cost = {
	get_incite_peasants_base_cost_var = {
		var = incite_peasants_base_cost_var
	}
	calculate_spy_action_cost_var = {
		spy_action_cost_var = incite_peasants_cost_var
		spy_action_base_cost = incite_peasants_base_cost_var
	}
}

#SCOPE = COUNTRY
calculate_stage_coup_cost = {
	get_stage_coup_base_cost_var = {
		var = stage_coup_base_cost_var
	}
	calculate_spy_action_cost_var = {
		spy_action_cost_var = stage_coup_cost_var
		spy_action_base_cost = stage_coup_base_cost_var
	}
}
