#SCOPE = COUNTRY
initialize_commune_integration_progress = {
	if = {
		limit = {
			NOT = {
				is_subject_of_type = integrated_commune
			}
		}
		set_variable = {
			which = commune_integration_progress
			value = 0
		}
	}
}

#SCOPE = COUNTRY
calculate_commune_integration_cost = {
	export_to_variable = {
		which = overlord_admin_efficiency
		value = modifier:administrative_efficiency
		who = overlord
	}
	set_variable = {
		overlord_admin_efficiency_multiplier = 1
	}
	subtract_variable = {
		which = overlord_admin_efficiency_multiplier
		which = overlord_admin_efficiency
	}
	if = {
		limit = {
			NOT = {
				check_variable = {
					overlord_admin_efficiency_multiplier = 0.1
				}
			}
		}
		set_variable = {
			overlord_admin_efficiency_multiplier = 0.1
		}
	}
	# Moldavian Commune and other variants will have steep penalties for diplo annexation cost.
	# This should not apply to commune subjects, just regular subjects, so we negate the penalty here.
	export_to_variable = {
		which = overlord_diplo_annex_multiplier
		value = modifier:diplomatic_annexation_cost
		who = overlord
	}
	if = {
		limit = {
			overlord = {
				has_reform = moldavian_commune_reform
			}
		}
		change_variable = {
			which = overlord_diplo_annex_multiplier
			value = -0.7
		}
	}
	change_variable = {
		which = overlord_diplo_annex_multiplier
		value = 1
	}
	if = {
		limit = {
			NOT = {
				check_variable = {
					overlord_diplo_annex_multiplier = 0.1
				}
			}
		}
		set_variable = {
			overlord_diplo_annex_multiplier = 0.1
		}
	}
	export_to_variable = {
		which = overlord_all_powers_cost
		value = modifier:all_power_cost
		who = overlord
	}
	change_variable = {
		which = overlord_all_powers_cost
		value = 1
	}
	# Subject type multiplier
	if = {
		limit = {
			is_subject_of_type = allied_commune
		}
		get_allied_commune_annexation_cost_multiplier = {
			var = subject_type_annexation_cost_multiplier
		}
	}
	else_if = {
		limit = {
			is_subject_of_type = sister_commune
		}
		get_sister_commune_annexation_cost_multiplier = {
			var = subject_type_annexation_cost_multiplier
		}
	}
	else = {
		get_subject_to_commune_annexation_cost_multiplier = {
			var = subject_type_annexation_cost_multiplier
		}
	}
	set_variable = {
		commune_integration_cost = 0
	}
	every_owned_province = {
		export_to_variable = {
			which = commune_integration_cost
			value = development
		}
		# === bring in out-of-scope variables from PREV ===
		set_variable = {
			which = subject_type_annexation_cost_multiplier
			which = PREV
		}
		set_variable = {
			which = overlord_diplo_annex_multiplier
			which = PREV
		}
		set_variable = {
			which = overlord_admin_efficiency_multiplier
			which = PREV
		}
		set_variable = {
			which = overlord_all_powers_cost
			which = PREV
		}
		# === apply multipliers ===
		multiply_variable = {
			which = commune_integration_cost
			which = subject_type_annexation_cost_multiplier
		}
		multiply_variable = {
			which = commune_integration_cost
			which = overlord_diplo_annex_multiplier
		}
		multiply_variable = {
			which = commune_integration_cost
			which = overlord_admin_efficiency_multiplier
		}
		multiply_variable = {
			which = commune_integration_cost
			which = overlord_all_powers_cost
		}
		# === local coring cost ===
		export_to_variable = {
			which = local_coring_cost
			value = modifier:local_core_creation
		}
		change_variable = {
			which = local_coring_cost
			value = 1
		}
		if = {
			limit = {
				NOT = {
					check_variable = {
						local_coring_cost = 0.2
					}
				}
			}
			set_variable = {
				local_coring_cost = 0.2
			}
		}
		multiply_variable = {
			which = commune_integration_cost
			which = local_coring_cost
		}
		# === push result back up to country ===
		PREV = {
			change_variable = {
				which = commune_integration_cost
				which = PREV
			}
		}
	}
	round_variable = {
		which = commune_integration_cost
		value = -1
	}
}

#SCOPE = COUNTRY
change_commune_integration_progress = {
	set_variable = {
		which = integration_progress_temp
		which = commune_integration_progress
	}
	change_variable = {
		which = integration_progress_temp
		value = $amount$
	}
	if = {
		limit = {
			NOT = {
				check_variable = {
					integration_progress_temp = 0
				}
			}
		}
		set_variable = {
			which = commune_integration_progress
			value = 0
		}
	}
	else_if = {
		limit = {
			check_variable = {
				which = integration_progress_temp
				which = commune_integration_cost
			}
		}
		set_variable = {
			which = commune_integration_progress
			which = commune_integration_cost
		}
	}
	else = {
		change_variable = {
			which = commune_integration_progress
			value = $amount$
		}
	}
}

#SCOPE = COUNTRY
calculate_monthly_commune_integration_progress = {
	if = {
		limit = {
			is_commune_subject = yes
		}
		set_variable = {
			which = peasant_zeal_amount
			value = 0
		}
		overlord = {
			export_government_power_to_variable = {
				mechanic_type = peasant_zealotry
				power_type = peasant_zeal
			}
			PREV = {
				change_variable = {
					which = peasant_zeal_amount
					which = PREV
				}
			}
		}
		set_variable = {
			which = commune_integration_progress_change
			which = peasant_zeal_amount
		}
		divide_variable = {
			which = commune_integration_progress_change
			value = 400
		}
	}
	else = {
		overlord = {
			calculate_monthly_subject_to_commune_integration_progress = yes
		}
	}
}

calculate_monthly_subject_to_commune_integration_progress = {
	get_base_dip_investment_monthly_cost = {
		var = subject_to_commune_monthly_change
	}
	export_to_variable = {
		which = diplo_rep
		value = modifier:diplomatic_reputation
	}
	variable_truncate_decimal = {
		var = diplo_rep
	}
	change_variable = {
		which = subject_to_commune_monthly_change
		which = diplo_rep
	}
	if = {
		limit = {
			NOT = {
				check_variable = {
					subject_to_commune_monthly_change = 1
				}
			}
		}
		set_variable = {
			subject_to_commune_monthly_change = 0
		}
	}
	else = {
		if = {
			limit = {
				dip_power = 0
			}
			export_to_variable = {
				which = dip_power_var
				value = trigger_value:dip_power
			}
		}
		else = {
			set_variable = {
				dip_power_var = 0
			}
		}
		#If you don't have enough dip points for monthly cost, the amount deducted will be what you have.
		if = {
			limit = {
				check_variable = {
					which = subject_to_commune_monthly_change
					which = dip_power_var
				}
			}
			set_variable = {
				which = subject_to_commune_monthly_change
				which = dip_power_var
			}
		}
	}
	if = {
		limit = {
			check_variable = {
				subject_to_commune_monthly_change = 1
			}
		}
		set_variable = {
			which = commune_integration_progress_change
			which = subject_to_commune_monthly_change
		}
		event_target:commune_being_integrated = {
			set_variable = {
				which = commune_integration_progress_change
				which = PREV
			}
		}
	}
}

#SCOPE = COUNTRY
calculate_overlord_adm_investment_integration_progress_gain = {
	if = {
		limit = {
			is_investing_adm_power_in_commune = yes
		}
		get_base_adm_investment_monthly_cost = {
			var = commune_adm_investment_monthly_cost
		}
		export_to_variable = {
			which = adm_eff
			value = modifier:administrative_efficiency
		}
		#+1/-1 adm spent per month for every 10% admin efficiency above or below 0.
		multiply_variable = {
			adm_eff = 10
		}
		round_variable = {
			which = adm_eff
			value = -1
		}
		change_variable = {
			which = commune_adm_investment_monthly_cost
			which = adm_eff
		}
		if = {
			limit = {
				NOT = {
					check_variable = {
						commune_adm_investment_monthly_cost = 1
					}
				}
			}
			set_variable = {
				commune_adm_investment_monthly_cost = 0
			}
		}
		else = {
			if = {
				limit = {
					adm_power = 0
				}
				export_to_variable = {
					which = adm_power_var
					value = trigger_value:adm_power
				}
			}
			else = {
				set_variable = {
					adm_power_var = 0
				}
			}
			#If you don't have enough adm points for monthly cost, the amount deducted will be what you have.
			if = {
				limit = {
					check_variable = {
						which = commune_adm_investment_monthly_cost
						which = adm_power_var
					}
				}
				set_variable = {
					which = commune_adm_investment_monthly_cost
					which = adm_power_var
				}
			}
		}
		if = {
			limit = {
				check_variable = {
					commune_adm_investment_monthly_cost = 1
				}
			}
			set_variable = {
				which = progress_from_adm_investment
				which = commune_adm_investment_monthly_cost
			}
			get_adm_to_integration_progress_exchange_rate = {
				var = adm_to_integration_progress_exchange_rate
			}
			multiply_variable = {
				which = progress_from_adm_investment
				which = adm_to_integration_progress_exchange_rate
			}
			set_variable = {
				which = commune_integration_progress_change
				which = progress_from_adm_investment
			}
			event_target:commune_being_integrated = {
				change_variable = {
					which = commune_integration_progress_change
					which = PREV
				}
			}
		}
	}
}
