#SCOPE = COUNTRY
initialize_commune_subject = {
	initialize_commune_integration_progress = yes
	calculate_commune_integration_cost = yes
}

#SCOPE = COUNTRY
initialize_commune_integration_progress = {
	if = {
		limit = {
			NOT = {
				is_subject_of_type = integrated_commune
			}
		}
		set_variable = {
			which = commune_integration_progress
			value = 0
		}
	}
}

#SCOPE = COUNTRY
calculate_every_commune_subject_integration_cost = {
	if = {
		limit = {
			has_government_mechanic = peasant_zealotry
		}
		every_subject_country = {
			limit = {
				NOT = {
					is_subject_of_type = integrated_commune
				}
			}
			calculate_commune_integration_cost = yes
		}
	}
}

#SCOPE = COUNTRY
calculate_commune_integration_cost = {
	export_to_variable = {
		which = overlord_admin_efficiency
		value = modifier:administrative_efficiency
		who = overlord
	}
	set_variable = {
		overlord_admin_efficiency_multiplier = 1
	}
	subtract_variable = {
		which = overlord_admin_efficiency_multiplier
		which = overlord_admin_efficiency
	}
	if = {
		limit = {
			NOT = {
				check_variable = {
					overlord_admin_efficiency_multiplier = 0.1
				}
			}
		}
		set_variable = {
			overlord_admin_efficiency_multiplier = 0.1
		}
	}
	# Moldavian Commune and other variants will have steep penalties for diplo annexation cost.
	# This should not apply to commune subjects, just regular subjects, so we negate the penalty here.
	export_to_variable = {
		which = overlord_diplo_annex_multiplier
		value = modifier:diplomatic_annexation_cost
		who = overlord
	}
	if = {
		limit = {
			overlord = {
				has_reform = moldavian_commune_reform
			}
		}
		change_variable = {
			which = overlord_diplo_annex_multiplier
			value = -0.7
		}
	}
	change_variable = {
		which = overlord_diplo_annex_multiplier
		value = 1
	}
	if = {
		limit = {
			NOT = {
				check_variable = {
					overlord_diplo_annex_multiplier = 0.1
				}
			}
		}
		set_variable = {
			overlord_diplo_annex_multiplier = 0.1
		}
	}
	export_to_variable = {
		which = overlord_all_powers_cost
		value = modifier:all_power_cost
		who = overlord
	}
	change_variable = {
		which = overlord_all_powers_cost
		value = 1
	}
	# Subject type multiplier
	if = {
		limit = {
			is_subject_of_type = allied_commune
		}
		set_variable = {
			subject_type_annexation_cost_multiplier = 3
		}
	}
	else_if = {
		limit = {
			is_subject_of_type = sister_commune
		}
		set_variable = {
			subject_type_annexation_cost_multiplier = 5
		}
	}
	else = {
		set_variable = {
			subject_type_annexation_cost_multiplier = 6
		}
	}
	set_variable = {
		commune_integration_cost = 0
	}
	every_owned_province = {
		export_to_variable = {
			which = commune_integration_cost
			value = development
		}
		# === bring in out-of-scope variables from PREV ===
		set_variable = {
			which = subject_type_annexation_cost_multiplier
			which = PREV
		}
		set_variable = {
			which = overlord_diplo_annex_multiplier
			which = PREV
		}
		set_variable = {
			which = overlord_admin_efficiency_multiplier
			which = PREV
		}
		set_variable = {
			which = overlord_all_powers_cost
			which = PREV
		}
		# === apply multipliers ===
		multiply_variable = {
			which = commune_integration_cost
			which = subject_type_annexation_cost_multiplier
		}
		multiply_variable = {
			which = commune_integration_cost
			which = overlord_diplo_annex_multiplier
		}
		multiply_variable = {
			which = commune_integration_cost
			which = overlord_admin_efficiency_multiplier
		}
		multiply_variable = {
			which = commune_integration_cost
			which = overlord_all_powers_cost
		}
		# === local coring cost ===
		export_to_variable = {
			which = local_coring_cost
			value = modifier:local_core_creation
		}
		change_variable = {
			which = local_coring_cost
			value = 1
		}
		if = {
			limit = {
				NOT = {
					check_variable = {
						local_coring_cost = 0.2
					}
				}
			}
			set_variable = {
				local_coring_cost = 0.2
			}
		}
		multiply_variable = {
			which = commune_integration_cost
			which = local_coring_cost
		}
		# === push result back up to country ===
		PREV = {
			change_variable = {
				which = commune_integration_cost
				which = PREV
			}
		}
	}
	round_variable = {
		which = commune_integration_cost
		value = -1
	}
}

#SCOPE = COUNTRY
change_commune_integration_progress = {
	set_variable = {
		which = integration_progress_temp
		which = commune_integration_progress
	}
	change_variable = {
		which = integration_progress_temp
		value = $amount$
	}
	if = {
		limit = {
			NOT = {
				check_variable = {
					integration_progress_temp = 0
				}
			}
		}
		set_variable = {
			which = commune_integration_progress
			value = 0
		}
	}
	else_if = {
		limit = {
			check_variable = {
				which = integration_progress_temp
				which = commune_integration_cost
			}
		}
		set_variable = {
			which = commune_integration_progress
			which = commune_integration_cost
		}
	}
	else = {
		change_variable = {
			which = commune_integration_progress
			value = $amount$
		}
	}
}

#SCOPE = COUNTRY
variable_change_commune_integration_progress = {
	set_variable = {
		which = integration_progress_temp
		which = commune_integration_progress
	}
	change_variable = {
		which = integration_progress_temp
		which = $var$
	}
	if = {
		limit = {
			NOT = {
				check_variable = {
					integration_progress_temp = 0
				}
			}
		}
		set_variable = {
			which = commune_integration_progress
			value = 0
		}
	}
	else_if = {
		limit = {
			check_variable = {
				which = integration_progress_temp
				which = commune_integration_cost
			}
		}
		set_variable = {
			which = commune_integration_progress
			which = commune_integration_cost
		}
	}
	else = {
		change_variable = {
			which = commune_integration_progress
			which = $var$
		}
	}
}

#SCOPE = COUNTRY
calculate_monthly_commune_integration_progress = {
	if = {
		limit = {
			is_commune_subject_being_integrated = yes
		}
		set_variable = {
			which = peasant_zeal_amount
			value = 0
		}
		overlord = {
			export_government_power_to_variable = {
				mechanic_type = peasant_zealotry
				power_type = peasant_zeal
			}
			PREV = {
				change_variable = {
					which = peasant_zeal_amount
					which = PREV
				}
			}
		}
		set_variable = {
			which = integration_progress_change
			which = peasant_zeal_amount
		}
		divide_variable = {
			which = integration_progress_change
			value = 400
		}
		variable_change_commune_integration_progress = {
			var = integration_progress_change
		}
	}
}

initialize_integration_progress_percentage_variable = {
	set_variable = {
		which = commune_integration_progress_percentage
		value = 0
	}
}

#SCOPE = COUNTRY
calculate_integration_progress_percentage = {
	if = {
		limit = {
			is_commune_subject_being_integrated = yes
		}
		set_variable = {
			which = commune_integration_progress_percentage
			which = commune_integration_progress
		}
		divide_variable = {
			which = commune_integration_progress_percentage
			which = commune_integration_cost
		}
		multiply_variable = {
			which = commune_integration_progress_percentage
			value = 100
		}
		round_variable = {
			which = commune_integration_progress_percentage
			value = 0
		}
	}
}
