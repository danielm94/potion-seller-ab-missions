calculate_commune_integration_cost = {
	export_to_variable = {
		which = overlord_admin_efficiency
		value = modifier:administrative_efficiency
		who = overlord
	}
	set_variable = {
		overlord_admin_efficiency_multiplier = 1
	}
	subtract_variable = {
		which = overlord_admin_efficiency_multiplier
		which = overlord_admin_efficiency
	}
	if = {
		limit = {
			NOT = {
				check_variable = {
					overlord_admin_efficiency_multiplier = 0.1
				}
			}
		}
		set_variable = {
			overlord_admin_efficiency_multiplier = 0.1
		}
	}
	# Moldavian Commune and other variants will have steep penalties for diplo annexation cost.
	# This should not apply to commune subjects, just regular subjects, so we negate the penalty here.
	export_to_variable = {
		which = overlord_diplo_annex_multiplier
		value = modifier:diplomatic_annexation_cost
		who = overlord
	}
	if = {
		limit = {
			overlord = {
				OR = {
					has_reform = moldavian_commune_reform
					has_reform = dacian_commune_reform
				}
			}
		}
		change_variable = {
			which = overlord_diplo_annex_multiplier
			value = -0.7
		}
	}
	change_variable = {
		which = overlord_diplo_annex_multiplier
		value = 1
	}
	if = {
		limit = {
			NOT = {
				check_variable = {
					overlord_diplo_annex_multiplier = 0.1
				}
			}
		}
		set_variable = {
			overlord_diplo_annex_multiplier = 0.1
		}
	}
	export_to_variable = {
		which = overlord_all_powers_cost
		value = modifier:all_power_cost
		who = overlord
	}
	change_variable = {
		which = overlord_all_powers_cost
		value = 1
	}
	# Subject type multiplier
	if = {
		limit = {
			is_subject_of_type = allied_commune
		}
		get_allied_commune_annexation_cost_multiplier = {
			var = subject_type_annexation_cost_multiplier
		}
	}
	else_if = {
		limit = {
			is_subject_of_type = sister_commune
		}
		if = {
			limit = {
				overlord = {
					has_country_flag = reduce_sister_commune_annexation_cost_multiplier
				}
			}
			get_discounted_sister_commune_annexation_cost_multiplier = {
				var = subject_type_annexation_cost_multiplier
			}
		}
		else = {
			get_sister_commune_annexation_cost_multiplier = {
				var = subject_type_annexation_cost_multiplier
			}
		}
	}
	else_if = {
		limit = {
			is_subject_of_type = consulate_subject
		}
		get_consulate_annexation_cost_multiplier = {
			var = subject_type_annexation_cost_multiplier
		}
	}
	else = {
		get_subject_to_commune_annexation_cost_multiplier = {
			var = subject_type_annexation_cost_multiplier
		}
	}
	set_variable = {
		commune_integration_cost = 0
	}
	every_owned_province = {
		export_to_variable = {
			which = commune_integration_cost
			value = development
		}
		# === bring in out-of-scope variables from PREV ===
		set_variable = {
			which = subject_type_annexation_cost_multiplier
			which = PREV
		}
		set_variable = {
			which = overlord_diplo_annex_multiplier
			which = PREV
		}
		set_variable = {
			which = overlord_admin_efficiency_multiplier
			which = PREV
		}
		set_variable = {
			which = overlord_all_powers_cost
			which = PREV
		}
		# === apply multipliers ===
		multiply_variable = {
			which = commune_integration_cost
			which = subject_type_annexation_cost_multiplier
		}
		multiply_variable = {
			which = commune_integration_cost
			which = overlord_diplo_annex_multiplier
		}
		multiply_variable = {
			which = commune_integration_cost
			which = overlord_admin_efficiency_multiplier
		}
		multiply_variable = {
			which = commune_integration_cost
			which = overlord_all_powers_cost
		}
		# === local coring cost ===
		export_to_variable = {
			which = local_coring_cost
			value = modifier:local_core_creation
		}
		change_variable = {
			which = local_coring_cost
			value = 1
		}
		if = {
			limit = {
				NOT = {
					check_variable = {
						local_coring_cost = 0.2
					}
				}
			}
			set_variable = {
				local_coring_cost = 0.2
			}
		}
		multiply_variable = {
			which = commune_integration_cost
			which = local_coring_cost
		}
		# === push result back up to country ===
		PREV = {
			change_variable = {
				which = commune_integration_cost
				which = PREV
			}
		}
	}
	round_variable = {
		which = commune_integration_cost
		value = -1
	}
}

change_commune_integration_progress = {
	set_variable = {
		which = integration_progress_temp
		which = commune_integration_progress
	}
	change_variable = {
		which = integration_progress_temp
		value = $amount$
	}
	if = {
		limit = {
			NOT = {
				check_variable = {
					integration_progress_temp = 0
				}
			}
		}
		set_variable = {
			which = commune_integration_progress
			value = 0
		}
	}
	else_if = {
		limit = {
			check_variable = {
				which = integration_progress_temp
				which = commune_integration_cost
			}
		}
		set_variable = {
			which = commune_integration_progress
			which = commune_integration_cost
		}
	}
	else = {
		change_variable = {
			which = commune_integration_progress
			value = $amount$
		}
	}
}

#SCOPE = COUNTRY
calculate_monthly_commune_integration_progress = {
	if = {
		limit = {
			is_commune_subject = yes
		}
		set_variable = {
			which = peasant_zeal_amount
			value = 0
		}
		overlord = {
			export_government_power_to_variable = {
				mechanic_type = peasant_zealotry
				power_type = peasant_zeal
			}
			PREV = {
				change_variable = {
					which = peasant_zeal_amount
					which = PREV
				}
			}
		}
		set_variable = {
			which = commune_integration_progress_change
			which = peasant_zeal_amount
		}
		divide_variable = {
			which = commune_integration_progress_change
			value = 400
		}
		if = {
			limit = {
				overlord = {
					has_country_flag = base_monthly_commune_intergation_rate_increase
				}
			}
			change_variable = {
				commune_integration_progress_change = 0.5
			}
		}
	}
	else = {
		overlord = {
			calculate_monthly_subject_to_commune_integration_progress = yes
		}
	}
}

calculate_monthly_subject_to_commune_integration_progress = {
	get_base_dip_investment_monthly_cost = {
		var = subject_to_commune_monthly_change
	}
	export_to_variable = {
		which = diplo_rep
		value = modifier:diplomatic_reputation
	}
	variable_truncate_decimal = {
		var = diplo_rep
	}
	change_variable = {
		which = subject_to_commune_monthly_change
		which = diplo_rep
	}
	if = {
		limit = {
			NOT = {
				check_variable = {
					subject_to_commune_monthly_change = 1
				}
			}
		}
		set_variable = {
			subject_to_commune_monthly_change = 0
		}
	}
	else = {
		if = {
			limit = {
				dip_power = 0
			}
			export_to_variable = {
				which = dip_power_var
				value = trigger_value:dip_power
			}
		}
		else = {
			set_variable = {
				dip_power_var = 0
			}
		}
		#If you don't have enough dip points for monthly cost, the amount deducted will be what you have.
		if = {
			limit = {
				check_variable = {
					which = subject_to_commune_monthly_change
					which = dip_power_var
				}
			}
			set_variable = {
				which = subject_to_commune_monthly_change
				which = dip_power_var
			}
		}
	}
	if = {
		limit = {
			check_variable = {
				subject_to_commune_monthly_change = 1
			}
		}
		set_variable = {
			which = commune_integration_progress_change
			which = subject_to_commune_monthly_change
		}
		PREV = {
			set_variable = {
				which = commune_integration_progress_change
				which = PREV
			}
		}
	}
}

#SCOPE = COUNTRY
calculate_overlord_adm_investment_integration_progress_gain = {
	if = {
		limit = {
			PREV = {
				is_investing_adm_power_in_commune = yes
			}
		}
		get_base_adm_investment_monthly_cost = {
			var = commune_adm_investment_monthly_cost
		}
		get_country_raw_governing_capacity = {
			var = gov_cap
		}
		export_to_variable = {
			which = gov_cap_mod
			value = modifier:governing_capacity_modifier
		}
		change_variable = {
			gov_cap_mod = 1
		}
		multiply_variable = {
			which = gov_cap
			which = gov_cap_mod
		}
		# +1 adm spent per month for every 250 gov cap
		divide_variable = {
			gov_cap = 250
		}
		round_variable = {
			which = gov_cap
			value = -1
		}
		change_variable = {
			which = commune_adm_investment_monthly_cost
			which = gov_cap
		}
		if = {
			limit = {
				NOT = {
					check_variable = {
						commune_adm_investment_monthly_cost = 1
					}
				}
			}
			set_variable = {
				commune_adm_investment_monthly_cost = 0
			}
		}
		else = {
			if = {
				limit = {
					adm_power = 0
				}
				export_to_variable = {
					which = adm_power_var
					value = trigger_value:adm_power
				}
			}
			else = {
				set_variable = {
					adm_power_var = 0
				}
			}
			#If you don't have enough adm points for monthly cost, the amount deducted will be what you have.
			if = {
				limit = {
					check_variable = {
						which = commune_adm_investment_monthly_cost
						which = adm_power_var
					}
				}
				set_variable = {
					which = commune_adm_investment_monthly_cost
					which = adm_power_var
				}
			}
		}
		if = {
			limit = {
				check_variable = {
					commune_adm_investment_monthly_cost = 1
				}
			}
			set_variable = {
				which = progress_from_adm_investment
				which = commune_adm_investment_monthly_cost
			}
			get_adm_to_integration_progress_exchange_rate = {
				var = adm_to_integration_progress_exchange_rate
			}
			multiply_variable = {
				which = progress_from_adm_investment
				which = adm_to_integration_progress_exchange_rate
			}
			set_variable = {
				which = commune_integration_progress_change
				which = progress_from_adm_investment
			}
			PREV = {
				change_variable = {
					which = commune_integration_progress_change
					which = PREV
				}
			}
		}
	}
}

release_balkanized_commune_country = {
	complex_release_effect = {
		target = $target$
		change_government = republic
		add_government_reform = peasants_republic
	}
	if = {
		limit = {
			$target$ = {
				has_country_flag = has_core_on_capital
			}
		}
		hidden_effect = {
			capital_scope = {
				add_core = $target$
			}
			$target$ = {
				clr_country_flag = has_core_on_capital
			}
		}
	}
	hidden_effect = {
		$target$ = {
			change_religion = capital
			if = {
				limit = {
					has_country_modifier = conversion_zeal
				}
				remove_country_modifier = conversion_zeal
				add_prestige = 100
				while = {
					limit = {
						NOT = {
							stability = 0
						}
					}
					add_stability = 1
				}
			}
		}
	}
	if = {
		limit = {
			exists = $target$
		}
		hidden_effect = {
			every_core_province = {
				limit = {
					owned_by = $target$
				}
				remove_core = PREV
			}
		}
		hidden_effect = {
			ROOT = {
				create_allied_commune = {
					who = $target$
				}
			}
			add_opinion = {
				who = ROOT
				modifier = balkanized_our_nation
			}
			add_liberty_desire = 5
			$target$ = {
				add_opinion = {
					who = ROOT
					modifier = balkanized_released_us
				}
				add_liberty_desire = -10
			}
		}
	}
}

andalusia_balkanization_galicia_effect = {
	event_target:new_peasant_subject = {
		if = {
			limit = {
				any_owned_province = {
					area = galicia_area
				}
				NOT = {
					exists = GAL
				}
			}
			custom_tooltip = andalusian_balkanization_GAL_tt
			hidden_effect = {
				every_owned_province = {
					limit = {
						area = galicia_area
					}
					remove_core = owner
					add_core = GAL
					cede_province = GAL
				}
			}
			if = {
				limit = {
					exists = GAL
				}
				hidden_effect = {
					root = {
						create_allied_commune = {
							who = GAL
						}
					}
					GAL = {
						change_religion = catholic
						change_primary_culture = galician
						change_government = republic
						add_government_reform = peasants_republic
						set_capital_if_owned = {
							id = 206
						}
					}
				}
			}
		}
	}
}

	set_variable = {

#SCOPE = COUNTRY (COMMUNE SUBJECT)
calculate_peasant_zeal_call_to_arms_cost = {
	if = {
		limit = {
			is_commune_subject = yes
		}
		set_variable = {
			num_wars_we_are_not_in = 0
		}
		save_event_target_as = this_subject
		overlord = {
			every_war_enemy_country = {
				limit = {
					NOT = {
						war_with = event_target:this_subject
					}
					OR = {
						is_in_war = {
							attacker_leader = this
							defender_leader = prev
						}
						is_in_war = {
							attacker_leader = prev
							defender_leader = this
						}
					}
				}
				event_target:this_subject = {
					change_variable = {
						num_wars_we_are_not_in = 1
					}
				}
			}
		}
		if = {
			limit = {
				is_subject_of_type = allied_commune
			}
			get_allied_commune_join_war_zeal_cost = {
				var = base_cost
			}
		}
		else_if = {
			limit = {
				is_subject_of_type = sister_commune
			}
			get_sister_commune_join_war_zeal_cost = {
				var = base_cost
			}
		}
		else = {
			get_integrated_commune_join_war_zeal_cost = {
				var = base_cost
			}
		}
		set_variable = {
			which = join_wars_peasant_zeal_cost
			which = base_cost
		}
		multiply_variable = {
			which = join_wars_peasant_zeal_cost
			which = num_wars_we_are_not_in
		}
		if = {
			limit = {
				has_saved_global_event_target = highest_coa_cost
			}
			if = {
				limit = {
					check_variable = {
						which = join_wars_peasant_zeal_cost
						which = event_target:highest_coa_cost
					}
				}
				save_global_event_target_as = highest_coa_cost
			}
		}
		else = {
			save_global_event_target_as = highest_coa_cost
		}
	}
}

call_all_peasant_subjects_to_arms_effect = {
	set_variable = {
		peasant_zeal_spent = 0
	}
	export_government_power_to_variable = {
		mechanic_type = peasant_zealotry
		power_type = peasant_zeal
	}
	while = {
		limit = {
			NOT = {
				has_country_flag = abort_peasant_call_to_arms_loop
			}
			check_variable = {
				peasant_zeal_amount = 1
			}
			is_variable_less_than_or_equal_to = {
				var = peasant_zeal_spent
				which = peasant_call_to_arms_zeal_budget
			}
			any_subject_country = {
				NOT = {
					liberty_desire = 50
				}
				is_commune_subject = yes
				if = {
					limit = {
						PREV = {
							NOT = {
								has_country_flag = cta_allied_communes
							}
						}
					}
					NOT = {
						is_subject_of_type = allied_commune
					}
				}
				if = {
					limit = {
						PREV = {
							NOT = {
								has_country_flag = cta_sister_communes
							}
						}
					}
					NOT = {
						is_subject_of_type = sister_commune
					}
				}
				if = {
					limit = {
						PREV = {
							NOT = {
								has_country_flag = cta_integrated_communes
							}
						}
					}
					NOT = {
						is_subject_of_type = integrated_commune
					}
				}
				NOT = {
					war_with = ROOT
				}
				has_peasant_zeal_cost_less_or_equal_to_budget = yes
			}
		}
		if = {
			limit = {
				has_country_flag = peasant_call_to_arms_strongest_military_strategy
			}
			call_strongest_commune_subject_effect = yes
		}
		else_if = {
			limit = {
				has_country_flag = peasant_call_to_arms_closest_to_target_strategy
			}
			call_nearest_commune_subject_effect = yes
		}
	}
	clr_country_flag = abort_peasant_call_to_arms_loop
}

call_strongest_commune_subject_effect = {
	every_subject_country = {
		limit = {
			NOT = {
				liberty_desire = 50
			}
			is_commune_subject = yes
			if = {
				limit = {
					PREV = {
						NOT = {
							has_country_flag = cta_allied_communes
						}
					}
				}
				NOT = {
					is_subject_of_type = allied_commune
				}
			}
			if = {
				limit = {
					PREV = {
						NOT = {
							has_country_flag = cta_sister_communes
						}
					}
				}
				NOT = {
					is_subject_of_type = sister_commune
				}
			}
			if = {
				limit = {
					PREV = {
						NOT = {
							has_country_flag = cta_integrated_communes
						}
					}
				}
				NOT = {
					is_subject_of_type = integrated_commune
				}
			}
			NOT = {
				war_with = ROOT
			}
		}
		if = {
			limit = {
				has_saved_global_event_target = strongest_commune_subject
			}
			if = {
				limit = {
					military_strength = {
						who = event_target:strongest_commune_subject
						value = 1.001
					}
				}
				save_global_event_target_as = strongest_commune_subject
			}
		}
		else = {
			save_global_event_target_as = strongest_commune_subject
		}
	}
	event_target:strongest_commune_subject = {
		set_variable = {
			which = peasant_zeal_spent
			which = overlord
		}
		change_variable = {
			which = peasant_zeal_spent
			which = join_wars_peasant_zeal_cost
		}
		set_variable = {
			which = peasant_zeal_amount
			which = overlord
		}
		subtract_variable = {
			which = peasant_zeal_amount
			which = join_wars_peasant_zeal_cost
		}
		set_variable = {
			which = peasant_call_to_arms_zeal_budget
			which = overlord
		}
		if = {
			limit = {
				is_variable_less_than_or_equal_to = {
					var = peasant_zeal_spent
					which = peasant_call_to_arms_zeal_budget
				}
				check_variable = {
					peasant_zeal_amount = 0
				}
			}
			join_all_offensive_wars_of = overlord
			join_all_defensive_wars_of = overlord
			PREV = {
				set_variable = {
					which = join_wars_peasant_zeal_cost
					which = event_target:strongest_commune_subject
				}
				change_variable = {
					which = peasant_zeal_spent
					which = join_wars_peasant_zeal_cost
				}
				multiply_variable = {
					join_wars_peasant_zeal_cost = -1
				}
				variable_government_power_effect = {
					var = join_wars_peasant_zeal_cost
					mechanic_type = peasant_zealotry
					power_type = peasant_zeal
					scaled_with_gain_modifier = no
				}
			}
		}
		else = {
			PREV = {
				set_country_flag = abort_peasant_call_to_arms_loop
			}
		}
	}
	clear_global_event_target = strongest_commune_subject
}
