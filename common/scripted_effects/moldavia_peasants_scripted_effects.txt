export_egyptian_grain_cost_effect = {
	add_years_of_income = -0.05
	if = {
		limit = {
			any_owned_province = {
				region = egypt_region
				trade_goods = grain
				base_production = 2
			}
		}
		random_owned_province = {
			limit = {
				region = egypt_region
				trade_goods = grain
				base_production = 2
			}
			add_base_production = -1
		}
	}
	else = {
		export_to_variable = {
			which = subject_grain_payment
			value = treasury
		}
		add_years_of_income = -0.1
		export_to_variable = {
			which = post_treasury
			value = treasury
		}
		subtract_variable = {
			which = subject_grain_payment
			which = post_treasury
		}
		random_province = {
			limit = {
				region = egypt_region
				trade_goods = grain
				base_production = 2
				owner = {
					is_subject_of_type_with_overlord = {
						who = root
						type = integrated_commune
					}
				}
			}
			add_base_production = -1
			owner = {
				set_variable = {
					which = subject_grain_payment
					which = root
				}
				variable_effect = {
					var = subject_grain_payment
					effect = add_treasury
				}
			}
		}
	}
}

establish_commune_effect = {
	$province$ = {
		custom_tooltip = establish_commune_effect_tt
		hidden_effect = {
			add_core = root
			remove_core = owner
			cede_province = root
		}
		[[center_of_trade]
		center_of_trade = $center_of_trade$
		]
		[[custom_effect]
		$custom_effect$
		]
		add_province_modifier = {
			name = $modifier$
			duration = -1
		}
	}
}

}

#SCOPE = PROVINCE
commune_transfer_province_or_mark_province_for_transfer = {
	negate_commune_coring_penalty = yes
	add_permanent_claim = root
	if = {
		limit = {
			owner = {
				is_subject_of_type_with_overlord = {
					who = ROOT
					type = integrated_commune
				}
			}
		}
		add_core = root
		remove_core = owner
		cede_province = root
	}
}
