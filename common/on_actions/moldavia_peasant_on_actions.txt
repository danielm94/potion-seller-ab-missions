on_province_owner_change = {
	if = {
		limit = {
			owner = {
				has_country_flag = mop_no_coring_penalty_in_perma_claim_provinces
			}
			is_permanent_claim = owner
		}
		if = {
			limit = {
				owner = {
					OR = {
						has_reform = moldavian_commune_reform
						has_reform = dacian_commune_reform
					}
				}
			}
			add_province_modifier = {
				name = negate_moldavian_commune_reform_ccr
				duration = -1
				hidden = yes
			}
		}
	}
}

# ROOT = PROVINCE YOU 
on_province_owner_change = {
	if = {
		limit = {
			NOT = {
				has_global_flag = set_province_peasant_modifiers
			}
			OR = {
				owner = {
					is_country_that_spreads_radicalized_peasants = yes
				}
				any_neighbor_province = {
					owner = {
						is_country_that_spreads_radicalized_peasants = yes
					}
				}
			}
		}
		set_global_flag = set_province_peasant_modifiers
		owner = {
			country_event = {
				id = moldavia_peasant_events.0
				days = 1
			}
		}
	}
}

on_dependency_gained = {
	FROM = {
		if = {
			limit = {
				OR = {
					is_country_that_spreads_radicalized_peasants = yes
				}
			}
			mark_country_has_been_spreader_of_radicalized_peasants = yes
			apply_radicalized_peasantry_effect = yes
		}
		if = {
			limit = {
				OR = {
					is_country_that_spreads_radicalized_peasants = yes
					has_nobility_estate_of_some_kind = no
				}
			}
			every_owned_province = {
				limit = {
					has_province_modifier = radicalized_peasantry
				}
				remove_province_modifier = radicalized_peasantry
			}
		}
	}
}

on_dependency_lost = {
	FROM = {
		if = {
			limit = {
				has_been_spreader_of_radicalized_peasants = yes
			}
			apply_radicalized_peasantry_effect = yes
		}
	}
}

on_siege_won_province = {
	if = {
		limit = {
			controlled_by = REB
			has_rebel_faction = anti_tax_rebels
			owner = {
				is_in_war = {
					casus_belli = cb_great_peasant_uprising
					defender_leader = THIS
				}
			}
		}
		random_country = {
			limit = {
				is_in_war = {
					attacker_leader = this
					defender_leader = FROM
					casus_belli = cb_great_peasant_uprising
				}
			}
			ROOT = {
				change_controller = PREV
			}
		}
	}
}
