on_monthly_pulse = {
	if = {
		limit = {
			ai = no
		}
		calculate_spy_action_cost_values = yes
	}
}

on_monthly_pulse = {
	if = {
		limit = {
			has_government_mechanic = peasant_zealotry
		}
		export_government_power_to_variable = {
			mechanic_type = peasant_zealotry
			power_type = peasant_zeal
		}
	}
}

on_startup = {
	if = {
		limit = {
			ai = no
		}
		calculate_spy_action_cost_values = yes
	}
}

on_startup = {
	if = {
		limit = {
			has_government_mechanic = peasant_zealotry
		}
		export_government_power_to_variable = {
			mechanic_type = peasant_zealotry
			power_type = peasant_zeal
		}
	}
}

on_province_owner_change = {
	if = {
		limit = {
			owner = {
				has_country_flag = mop_no_coring_penalty_in_perma_claim_provinces
			}
			is_permanent_claim = owner
		}
		if = {
			limit = {
				owner = {
					has_reform = moldavian_commune_reform
				}
			}
			add_province_modifier = {
				name = negate_moldavian_commune_reform_ccr
				duration = -1
				hidden = yes
			}
		}
	}
}
