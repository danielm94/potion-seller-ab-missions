			}

		}
	}

# ROOT = PROVINCE YOU 
on_province_owner_change = {
	if = {
		limit = {
			NOT = {
				has_global_flag = set_province_peasant_modifiers
			}
			OR = {
				owner = {
					is_country_that_spreads_radicalized_peasants = yes
				}
				any_neighbor_province = {
					owner = {
						is_country_that_spreads_radicalized_peasants = yes
					}
				}
			}
		}
		set_global_flag = set_province_peasant_modifiers
		owner = {
			country_event = {
				id = moldavia_peasant_events.0
				days = 1
			}
		}
	}
}

on_dependency_gained = {
	FROM = {
		if = {
			limit = {
				OR = {
					is_country_that_spreads_radicalized_peasants = yes
				}
			}
			mark_country_has_been_spreader_of_radicalized_peasants = yes
			if = {
				limit = {
					NOT = {
						has_global_flag = set_province_peasant_modifiers
					}
				}
				set_global_flag = set_province_peasant_modifiers
				ROOT = {
					country_event = {
						id = moldavia_peasant_events.0
						days = 1
					}
				}
			}
		}
		if = {
			limit = {
				OR = {
					is_country_that_spreads_radicalized_peasants = yes
					has_nobility_estate_of_some_kind = no
				}
			}
			every_owned_province = {
				limit = {
					has_radicalized_peasantry_province = yes
				}
				remove_province_modifier = radicalized_peasantry
			}
		}
	}
}

on_dependency_lost = {
	FROM = {
		if = {
			limit = {
				has_been_spreader_of_radicalized_peasants = yes
			}
			apply_radicalized_peasantry_effect = yes
		}
	}
}
