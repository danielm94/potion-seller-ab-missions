on_province_owner_change = {
	if = {
		limit = {
			owner = {
				has_country_flag = mop_no_coring_penalty_in_perma_claim_provinces
			}
			is_permanent_claim = owner
		}
		if = {
			limit = {
				owner = {
					OR = {
						has_reform = moldavian_commune_reform
						has_reform = dacian_commune_reform
					}
				}
			}
			add_province_modifier = {
				name = negate_moldavian_commune_reform_ccr
				duration = -1
				hidden = yes
			}
		}
	}
}

on_dependency_lost = {
	if = {
		limit = {
			FROM = {
				is_or_was_tag = {
					tag = ZAZ
				}
			}
		}
		remove_country_modifier = mop_zaz_subject_bonus
	}
}

on_dependency_gained = {
	if = {
		limit = {
			has_country_flag = mop_zaz_subject_bonus_enabled
			FROM = {
				is_or_was_tag = {
					tag = ZAZ
				}
			}
		}
		add_country_modifier = {
			name = mop_zaz_subject_bonus
			duration = -1
		}
	}
}

on_monthly_pulse = {
	if = {
		limit = {
			has_country_flag = support_rebels_against_@THIS
			any_country = {
				has_spawned_supported_rebels = PREV
			}
		}
		clr_country_flag = support_rebels_against_@THIS
		random_country = {
			limit = {
				has_spawned_supported_rebels = PREV
			}
			clr_country_flag = support_rebels_against_@PREV
		}
	}
}
