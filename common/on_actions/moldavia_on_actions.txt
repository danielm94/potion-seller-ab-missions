on_monthly_pulse = {
	if = {
		limit = {
			ai = no
		}
		calculate_spy_action_cost_values = yes
	}
	calculate_every_commune_subject_integration_cost = yes
	calculate_monthly_commune_integration_progress_pulse_effect = yes
	calculate_integration_progress_percentage = yes
}

on_startup = {
	if = {
		limit = {
			ai = no
		}
		calculate_spy_action_cost_values = yes
	}
	calculate_every_commune_subject_integration_cost = yes
	calculate_monthly_commune_integration_progress = yes
	calculate_overlord_adm_investment_integration_progress_gain = yes
	calculate_integration_progress_percentage = yes
}

on_dependency_gained = {
	if = {
		limit = {
			FROM = {
				is_commune_subject = yes
			}
		}
		FROM = {
			initialize_commune_subject = yes
		}
	}
}
