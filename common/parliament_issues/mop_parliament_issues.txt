commune_integration_issue = {
	category = 2
	allow = {
		has_issue_unlocked = {
			issue = commune_integration_issue
		}
	}
	effect = {
		custom_tooltip = 25_percent_integration_progress_gain_tt
		hidden_effect = {
			every_subject_country = {
				limit = {
					is_commune_subject_being_integrated = yes
				}
				if = {
					limit = {
						NOT = {
							has_saved_global_event_target = largest_integration_cost
						}
					}
					save_global_event_target_as = largest_integration_cost
				}
				else = {
					if = {
						limit = {
							check_variable = {
								which = commune_integration_cost
								which = event_target:largest_integration_cost
							}
						}
						save_global_event_target_as = largest_integration_cost
					}
				}
				if = {
					limit = {
						has_saved_global_event_target = largest_integration_cost
					}
					event_target:largest_integration_cost = {
						change_commune_integration_progress_as_percentage_of_cost = {
							percentage = 0.25
						}
					}
				}
			}
		}
	}
	modifier = {
		diplomatic_annexation_cost = -0.1
		diplomatic_reputation = 1
		leaguists_faction_influence = 0.05
	}
	chance = {
		factor = 5
	}
	ai_will_do = {
		factor = 500
	}
}
