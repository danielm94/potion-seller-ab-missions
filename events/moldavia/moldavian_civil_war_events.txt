namespace = moldavian_civil_war_events
#Disaster Start - Voivod is killed by boyars
country_event = {
	id = moldavian_civil_war_events.1
	title = moldavian_civil_war_events.1.t
	desc = moldavian_civil_war_events.1.d
	picture = KING_ASSASSINATION_eventPicture
	is_triggered_only = yes
	immediate = {
		hidden_effect = {
			set_country_flag = started_moldavian_civil_war
			set_country_flag = in_moldavian_civil_war
			kill_ruler = yes
			setup_MOP_effect = yes
			setup_boyar_MOL_effect = yes
		}
	}
	option = {
		name = moldavian_civil_war_events.1.a
		add_ruler_personality = impaler_personality
		add_prestige = -25
		add_legitimacy = -50
		hidden_effect = {
			if = {
				limit = {
					NOT = {
						army_size_percentage = 0.2
					}
				}
				add_ruler_modifier = {
					name = moldavian_civil_war_anti_cheese
					duration = -1
				}
				capital_scope = {
					artillery = owner
					artillery = owner
					artillery = owner
				}
				define_advisor = {
					type = commandant
					cost_multiplier = 0.05
					skill = 3
					name = "Shouldn't Have Cheesed LMAO"
				}
			}
			if = {
				limit = {
					NOT = {
						army_size_percentage = 1
					}
				}
				while = {
					limit = {
						NOT = {
							army_size_percentage = 1
						}
					}
					capital_scope = {
						infantry = owner
					}
				}
			}
			if = {
				limit = {
					is_bankrupt = yes
				}
				remove_country_modifier = bankruptcy
				add_stability = 6
				add_war_exhaustion = -20
				add_legitimacy = 100
				add_prestige = 200
				add_adm_power = 200
				add_dip_power = 200
				add_mil_power = 200
			}
		}
		declare_war_with_cb = {
			casus_belli = cb_moldavia_civil_war
			who = MOP
		}
		switch_tag = MOP
	}
}

#Civil War End - Razesi Victory
country_event = {
	id = moldavian_civil_war_events.3
	title = moldavian_civil_war_events.3.t
	desc = moldavian_civil_war_events.3.d
	picture = GFX_moldavian_civil_war_events.3
	trigger = {
		tag = MOP
		OR = {
			has_won_moldavian_civil_war = yes
			NOT = {
				exists = MOL
			}
			NOT = {
				MOL = {
					any_owned_province = {
						OR = {
							region = carpathia_region
							region = balkan_region
						}
					}
				}
			}
		}
	}
	fire_only_once = yes
	immediate = {
		set_country_flag = moldavian_civil_war_victory
	}
	option = {
		name = moldavian_civil_war_events.3.a
		add_stability = 2
		add_ruler_personality = peasant_revolt_leader_personality
		add_government_reform = moldavian_commune_reform
		balance_peasant_faction_influence_effect = yes
		add_faction_influence = {
			faction = abolitionists_faction
			influence = 20
		}
		add_country_modifier = {
			name = moldavia_civil_war_recovery
			duration = 3650
		}
		swap_non_generic_missions = yes
		country_event = {
			id = moldavia_peasant_events.1
		}
	}
}
