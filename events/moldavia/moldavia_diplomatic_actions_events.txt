#Country Finished Integrating
country_event = {
	id = moldavia_diplomatic_actions_events.2
	title = moldavia_diplomatic_actions_events.2.t
	desc = moldavia_diplomatic_actions_events.2.d
	picture = DEBATE_REPUBLICAN_eventPicture
	trigger = {
		is_integrating_commune_subject = yes
		any_subject_country = {
			is_commune_subject_being_integrated = yes
			check_variable = {
				which = commune_integration_progress
				which = commune_integration_cost
			}
		}
	}
	immediate = {
		hidden_effect = {
			disable_commune_integration_overlord = yes
			clear_commune_integration_investments = yes
			event_target:commune_being_integrated = {
				disable_commune_integration_subject = yes
				set_country_flag = commune_integrated_recently
			}
			if = {
				limit = {
					event_target:commune_being_integrated = {
						is_subject_of_type = sister_commune
					}
				}
				create_subject = {
					subject_type = integrated_commune
					who = event_target:commune_being_integrated
				}
			}
			else = {
				create_subject = {
					subject_type = sister_commune
					who = event_target:commune_being_integrated
				}
			}
			clear_global_event_target = commune_being_integrated
		}
	}
	option = {
		name = moldavia_diplomatic_actions_events.2.a
		if = {
			limit = {
				NOT = {
					has_country_flag = no_integrate_penalty
				}
			}
			tooltip = {
				if = {
					limit = {
						event_target:commune_being_integrated = {
							is_subject_of_type = sister_commune
						}
					}
					create_subject = {
						subject_type = integrated_commune
						who = event_target:commune_being_integrated
					}
				}
				else = {
					create_subject = {
						subject_type = sister_commune
						who = event_target:commune_being_integrated
					}
				}
			}
			add_country_modifier = {
				name = "annexers_legacy"
				duration = 3650
			}
		}
	}
}
