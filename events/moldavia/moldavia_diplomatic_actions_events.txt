#Country Finished Integrating
country_event = {
	id = moldavia_diplomatic_actions_events.2
	title = moldavia_diplomatic_actions_events.2.t
	desc = moldavia_diplomatic_actions_events.2.d
	picture = DEBATE_REPUBLICAN_eventPicture
	is_triggered_only = yes
	immediate = {
		hidden_effect = {
			FROM = {
				clear_commune_integration_flags = yes
				disable_commune_integration_subject = yes
				set_country_flag = commune_integrated_recently
				clr_country_flag = notified_overlord_ready_to_integrate
			}
			if = {
				limit = {
					NOT = {
						any_subject_country = {
							is_commune_subject_being_integrated = yes
						}
					}
				}
				disable_commune_integration_overlord = yes
			}
			pop_commune_subject_investing_mana_modifier = yes
			if = {
				limit = {
					FROM = {
						is_subject_of_type = allied_commune
					}
				}
				create_subject = {
					subject_type = integrated_commune
					who = FROM
				}
				FROM = {
					every_owned_province = {
						limit = {
							is_permanent_claim = root
						}
						if = {
							limit = {
								is_core = from
							}
							remove_core = from
							add_core = root
						}
						cede_province = root
					}
					if = {
						limit = {
							has_country_flag = model_commune
						}
						clr_country_flag = model_commune
						reverse_remove_opinion = {
							who = root
							modifier = designated_us_as_model_comune
						}
						root = {
							remove_country_modifier = designated_model_commune
						}
					}
				}
			}
			else_if = {
				limit = {
					FROM = {
						is_subject_of_type = consulate_subject
					}
				}
				create_allied_commune = {
					who = FROM
				}
			}
			else = {
				create_subject = {
					subject_type = allied_commune
					who = FROM
				}
				FROM = {
					change_government = republic
					add_government_reform = peasants_republic
				}
			}
			calculate_total_development_of_commune_subjects = yes
			map_all_integrating_commune_subjects_to_slots = yes
		}
	}
	option = {
		name = moldavia_diplomatic_actions_events.2.a
		if = {
			limit = {
				NOT = {
					has_country_flag = no_integrate_penalty
				}
				NOT = {
					has_government_attribute = MAY_free_diplo_annexation
				}
				FROM = {
					NOT = {
						has_country_flag = remove_annexers_legacy_token
					}
				}
			}
			add_country_modifier = {
				name = "annexers_legacy"
				duration = 3650
			}
		}
		else = {
			FROM = {
				clr_country_flag = remove_annexers_legacy_token
			}
		}
		tooltip = {
			if = {
				limit = {
					FROM = {
						is_subject_of_type = integrated_commune
					}
				}
				custom_tooltip = gain_all_perma_claim_provinces_tt
				custom_tooltip = AB_EMPTY
				create_subject = {
					subject_type = integrated_commune
					who = FROM
				}
			}
			else = {
				create_allied_commune = {
					who = FROM
				}
			}
		}
		if = {
			limit = {
				has_country_flag = peasant_zeal_on_commune_integration
			}
			change_peasant_zeal = {
				value = 5
			}
		}
		add_faction_influence = {
			faction = leaguists_faction
			influence = 10
		}
	}
}

			NOT = {

			}

			}

			target = overlord
		}

		set_country_flag = notified_overlord_ready_to_integrate
		overlord = {
			country_event = {
				id = moldavia_diplomatic_actions_events.6				#Event that actually alerts the player about the subject being ready
			}
		}
	}
	option = {
		name = AB_EMPTY
	}
}
